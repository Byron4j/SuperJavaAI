```java
/**
 * 把整个模块代码喂给 LLM，让它帮你找出问题
 */
public class CodeAuditTool {

    private final AIClient aiClient;

    /**
     * 对给定类进行结构分析
     */
    public AuditReport auditClass(String className, String sourceCode) {
        String prompt = """
            你是 Java 代码审查专家。分析以下代码，输出 JSON：

            {
              "responsibility": "这个类的核心职责(一句话)",
              "problems": [
                { "severity": "CRITICAL|HIGH|MEDIUM|LOW",
                  "type": "GOD_CLASS|LONG_METHOD|TIGHT_COUPLING|...",
                  "description": "问题描述",
                  "suggestion": "重构建议" }
              ],
              "dependencies": ["依赖/被依赖的类名"],
              "testability": "EASY|MEDIUM|HARD",
              "refactoring_priority": 1-10
            }

            代码：
            %s
            """.formatted(sourceCode);

        String result = aiClient.chat(prompt);
        return parseAuditReport(result);
    }

    /**
     * 对整个模块进行批量审计
     */
    public ModuleAuditResult auditModule(String modulePath) {
        List<String> javaFiles = scanJavaFiles(modulePath);

        // 并发分析所有文件
        try (var scope = new StructuredTaskScope.ShutdownOnFailure()) {
            List<StructuredTaskScope.Subtask<AuditReport>> tasks = javaFiles.stream()
                .map(f -> scope.fork(() -> auditClass(f, readFile(f))))
                .toList();

            scope.join();

            return new ModuleAuditResult(
                tasks.stream().map(StructuredTaskScope.Subtask::get).toList()
            );
        } catch (Exception e) {
            throw new RuntimeException("Audit failed", e);
        }
    }
}
```

## 二、阶段 1：构建安全网——测试先行

### 2.1 AI 生成特征测试（Characterization Tests）

在没有单元测试的遗留系统中，**特征测试**是重构的前置条件。它不测试"正确性"，而是测试"当前行为"——哪怕是错误的行为，也要先记录下来。

```java
/**
 * AI 辅助生成特征测试
 * 原理：输入各种边界值，记录当前输出作为"基线"
 */
@Service
public class CharacterizationTestGenerator {

    private final AIClient aiClient;
    private final TestRunner testRunner;

    public String generateTest(String className, String sourceCode) {
        String prompt = """
            为以下 Java 类生成 JUnit 5 特征测试（Characterization Test）。
            特征测试的目标是记录当前行为，不是验证正确性。

            规则：
            1. 分析所有 public 方法
            2. 为每个方法生成 3-5 个测试用例（正常值+边界值+null）
            3. 测试方法以 @Test + @DisplayName 标注
            4. 使用 assertEquals 记录当前期望的输出值
            5. 对于依赖外部服务的代码，使用 Mockito mock
            6. 对于无法 mock 的 static 调用，用 @Disabled 标注并注释原因

            类名：%s
            源代码：
            %s
            """.formatted(className, sourceCode);

        return aiClient.chat(prompt);
    }
}
```

### 2.2 实战：AI 为遗留代码生成的测试示例

```java
// 原始遗留代码
public class PriceCalculator {

    public static double calculate(String userType, double amount, String couponCode) {
        // 一堆 if-else 嵌套的"意大利面条"
        if (userType == null || userType.isEmpty()) {
            return amount;
        }
        double discount = 1.0;
        if ("VIP".equals(userType)) {
            discount = 0.8;
        } else if ("SVIP".equals(userType)) {
            discount = 0.65;
        }
        if (couponCode != null && !couponCode.isEmpty()) {
            if ("SAVE10".equals(couponCode)) {
                discount -= 0.1;
            } else if ("SAVE20".equals(couponCode)) {
                discount -= 0.2;
            }
        }
        return amount * Math.max(discount, 0.3);
    }
}

// AI 生成的特征测试
class PriceCalculatorCharacterizationTest {

    @Test
    @DisplayName("null用户类型 → 返回原价")
    void calculateWithNullUserType() {
        double result = PriceCalculator.calculate(null, 100.0, null);
        assertEquals(100.0, result, 0.001);
    }

    @Test
    @DisplayName("VIP用户无优惠券 → 8折")
    void calculateVipNoCoupon() {
        double result = PriceCalculator.calculate("VIP", 100.0, null);
        assertEquals(80.0, result, 0.001);
    }

    @Test
    @DisplayName("SVIP用户 + SAVE20 → 叠加后4.5折")
    void calculateSvipWithSave20() {
        double result = PriceCalculator.calculate("SVIP", 100.0, "SAVE20");
        assertEquals(45.0, result, 0.001);
    }

    @Test
    @DisplayName("折后低于3折 → 保底3折")
    void calculateBelowMinimumDiscount() {
        double result = PriceCalculator.calculate("SVIP", 100.0, "SAVE20");
        // 0.65 - 0.20 = 0.45, 且 > 0.3, 不触发保底
        assertEquals(45.0, result, 0.001);
    }
}
```

## 四、阶段 3：执行重构——模式与实践

### 4.1 模式一：绞杀者模式（Strangler Fig Pattern）

不一次性重写，而是逐步用新服务替换旧功能。AI 帮你生成路由层：

```java
/**
 * 绞杀者路由：逐步从旧系统迁移到新系统
 */
@RestController
@RequestMapping("/api/orders")
public class OrderRouterController {

    private final LegacyOrderService legacyService;
    private final NewOrderService newService;
    private final FeatureFlagService featureFlag;

    @PostMapping
    public OrderResponse createOrder(@RequestBody CreateOrderRequest request) {
        // 根据用户ID哈希灰度，逐步切换
        int hash = Math.abs(request.getUserId().hashCode()) % 100;

        if (featureFlag.isEnabled("NEW_ORDER_SERVICE", request.getUserId())) {
            return newService.createOrder(request);
        } else {
            return legacyService.createOrder(request);
        }
    }
}
```

```java
// 清晰的分层架构
@RestController
@RequestMapping("/api/orders")
public class OrderController {

    private final OrderApplicationService orderApplicationService;

    @PostMapping
    public OrderResponse submit(@Valid @RequestBody CreateOrderRequest request) {
        return orderApplicationService.submit(request);
    }
}

@Service
public class OrderApplicationService {

    private final InventoryDomainService inventoryService;
    private final PricingDomainService pricingService;
    private final OrderRepository orderRepository;
    private final NotificationService notificationService;

    @Transactional
    public OrderResponse submit(CreateOrderRequest request) {
        // 清晰简洁的编排逻辑
        Inventory inventory = inventoryService.checkAndReserve(
            request.getProductId(), request.getQuantity());

        Money finalPrice = pricingService.calculate(
            request.getUserId(), request.getProductId(),
            request.getQuantity(), request.getCouponCode());

        Order order = Order.create(request, finalPrice);
        Order savedOrder = orderRepository.save(order);

        notificationService.sendOrderConfirmation(savedOrder);

        return OrderResponse.from(savedOrder);
    }
}
```

## 五、阶段 4：验收——证明重构没有搞坏系统

### 5.1 行为等价性验证

```java
/**
 * 并行运行新旧系统，对比输出
 */
@Service
public class BehavioralEquivalenceValidator {

    /**
     * 影子流量对比：同时调用新旧服务，对比结果
     */
    public ValidationReport compareOldAndNew(
            List<TestCase> testCases) {

        List<Discrepancy> discrepancies = new ArrayList<>();

        for (TestCase tc : testCases) {
            Result oldResult = callOldService(tc.input());
            Result newResult = callNewService(tc.input());

            if (!oldResult.equals(newResult)) {
                discrepancies.add(new Discrepancy(
                    tc, oldResult, newResult,
                    analyzeDiscrepancy(oldResult, newResult)));
            }
        }

        return new ValidationReport(discrepancies);
    }

    private String analyzeDiscrepancy(Result old, Result nevv) {
        // 用 AI 判断差异是否为 Bug 修复，还是引入了新问题
        return aiClient.chat("""
            旧系统输出：%s
            新系统输出：%s

            这个差异是 Bug 修复（旧系统是错的），
            还是新系统引入了回归？
            """.formatted(old, nevv));
    }
}
```

## 七、独特观点：重构的第一性原理

**重构的目标不是让代码变漂亮，而是让代码具备"可演化性"。**

什么是可演化性？当一个新人（或 AI）接手这个模块，他能在一小时内理解并做出安全修改，这个模块就是可演化的。

因此，重构评判标准应该是：
- ❌ 不应该是：圈复杂度从 35 降到 5
- ✅ 而应该是：新来的实习生能否在 30 分钟内完成一个简单需求

AI 辅助重构的本质，是用机器智能弥补人类对复杂系统的理解极限。

## 八、总结

遗留系统重构的核心方法论：

1. **评估先行**：用 AI + SonarQube 审计，识别热点和风险
2. **测试为基**：AI 生成特征测试，锚定当前行为
3. **绞杀者渐进**：不一次性重写，用路由层逐步替换
4. **影子验证**：新旧系统并行运行，行为等价才上线
5. **AI 提效**：提取方法、消除重复、生成测试——这些重复劳动交给 AI

10 万行代码不可怕。可怕的是没有方法论的蛮干。AI 让你的重构从"艺术"变成"工程"。
