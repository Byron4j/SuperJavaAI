```

  小王：确认地址 → 判断归属（环保/城管/公安？）
       → 填入工单系统 → 选择分类 → 撰写描述
       耗时：8分钟

  市民："XX路烧烤店扰民！"

       → 又创建了一个新工单
       耗时：7分钟

... 重复循环 ...

12:00 上午合计：处理23通电话
  - 其中6通是"重复投诉"（同一事件多人投诉）
  - 其中4通是可以"即问即答"不用建单的
```

### 3.1 智能诉求分类引擎

```java
@Service
public class AppealClassifier {
    
    private final ChatClient chatClient;
    private final CategoryTreeService categoryService;
    
    /**
     * 实时诉求分类
     * 
     * 国家12345分类标准：5大类，200+小类
     * 一级：咨询/投诉/举报/建议/求助
     * 二级：城乡建设/市场监管/交通运输/...
     * 三级：具体事项
     */
    public ClassificationResult classify(String transcript, String city) {
        
        // 加载该城市的分类树（不同城市可能有本地化分类）
        CategoryTree categoryTree = categoryService.getCityCategoryTree(city);
        
        // 先用规则快速判断（常用高频分类）
        Optional<ClassificationResult> quickResult = quickClassify(transcript);
        if (quickResult.isPresent() && quickResult.get().getConfidence() > 0.9) {
            return quickResult.get();
        }
        
        // LLM精确分类
        String classification = chatClient.prompt()
            .system("""
                你是%s12345热线分类专家。将市民诉求正确归类。
                
                分类标准（%s）：
                一级 > 二级 > 三级 > 四级
                
                分类原则：
                1. 按"事权归属"分类，不按"表达方式"分类
                2. 涉及多个部门的，按"主要诉求"分类
                3. 能精确到四级的不要停留在二级
                4. 确实无法归类的才选"其他"
                5. 同时给出2-3个可能的分类（供坐席选择）
                
                请输出JSON：
                {
                  "primaryCategory": {"level1": "投诉", "level2": "城市管理", "level3": "噪音扰民", "level4": "商业噪音"},
                  "alternativeCategories": [
                    {"level1": "投诉", "level2": "环境保护", "level3": "噪声污染", "level4": "社会生活噪声"}
                  ],
                  "confidence": 0.95,
                  "responsibleDepartment": "城管执法局",
                  "keywords": ["烧烤店", "夜间", "音乐", "扰民"],
                  "urgency": "NORMAL",  
                  "isRepeatOffender": false
                }
                """.formatted(city, categoryTree.getDescription()))
            .user("""
                市民来电内容：
                %s
                
                请分类。
                """.formatted(transcript))
            .call()
            .content();
        
        return parseClassification(classification);
    }
    
    /**
     * 规则快速分类
     */
    private Optional<ClassificationResult> quickClassify(String transcript) {
        
        // 高频关键词 → 直接映射
        Map<String, ClassificationResult> quickMap = new HashMap<>();
        
        // 噪音类
        if (transcript.matches(".*(噪音|扰民|太吵|声音大|施工|装修).*")) {
            if (transcript.contains("夜间")) {
                return Optional.of(ClassificationResult.quick(
                    "投诉", "城市管理", "噪音扰民", "夜间施工", 0.92
                ));
            }
            return Optional.of(ClassificationResult.quick(
                "投诉", "城市管理", "噪音扰民", "其他噪音", 0.88
            ));
        }
        
        // 物业类
        if (transcript.matches(".*(物业|小区|电梯|停车|漏水|垃圾).*")) {
            return Optional.of(ClassificationResult.quick(
                "投诉", "城乡建设", "物业管理", 
                determineSubCategory(transcript), 0.90
            ));
        }
        
        // 消费投诉
        if (transcript.matches(".*(买到|假货|退款|不退|质量问题|商家).*")) {
            return Optional.of(ClassificationResult.quick(
                "投诉", "市场监管", "消费维权", 
                "商品质量", 0.91
            ));
        }
        
        return Optional.empty();
    }
}
```

### 3.2 相似工单去重引擎

```java
@Service
public class WorkOrderDedupEngine {
    
    private final MilvusClient milvusClient;
    private final EmbeddingService embeddingService;
    private final WorkOrderRepository orderRepository;
    
    /**
     * 相似工单智能匹配
     * 
     * 目的：检测新来电工单是否与已有工单描述的是同一事件
     * 如果是 → 合并，避免多头派单
     */
    public DedupResult checkDuplicate(String transcript, 
                                       ClassificationResult classification) {
        
        // 1. 向量化当前诉求
        EmbeddingVector currentVector = embeddingService.embed(
            buildDedupText(transcript, classification)
        );
        
        // 2. 在同一分类+同一区域+近24小时内搜索相似工单
        List<SimilarWorkOrder> candidates = milvusClient.search(
            "work_order_embeddings",
            currentVector,
            10,
            buildFilter(classification)
        );
        
        // 3. 如果没有候选，肯定不是重复
        if (candidates.isEmpty()) {
            return DedupResult.notDuplicate();
        }
        
        // 4. LLM精细判断是否重复
        return preciseDedup(transcript, candidates, classification);
    }
    
    private DedupResult preciseDedup(String currentTranscript,
                                      List<SimilarWorkOrder> candidates,
                                      ClassificationResult classification) {
        
        String candidatesSummary = candidates.stream()
            .map(wo -> String.format(
                "工单号：%s | 时间：%s | 地址：%s | 内容：%s | 状态：%s",
                wo.getOrderNumber(),
                wo.getCreateTime(),
                wo.getAddress(),
                wo.getDescription(),
                wo.getStatus()
            ))
            .collect(Collectors.joining("\n---\n"));
        
        String dedupJson = chatClient.prompt()
            .system("""
                你是12345热线工单审核员。判断新来电工单是否与已有工单是同一事件。
                
                判断标准：
                1. 同一事件：同一地址+同一问题+同一投诉对象 → 是重复（合并）
                2. 同类事件：同一问题但不同地址/对象 → 不是重复（新建）
                3. 关联事件：不同问题但是同一根因 → 建议关联（可批次处理）
                
                输出JSON：
                {
                  "isDuplicate": true/false,
                  "matchedOrderNumber": "匹配的工单号（如有）",
                  "matchReason": "匹配原因",
                  "suggestion": "建议合并/建议关联/建议新建"
                }
                """)
            .user("""
                新来电内容：%s
                新来电地址：%s
                
                待匹配的相似工单：
                %s
                """.formatted(
                    currentTranscript,
                    classification.getAddress(),
                    candidatesSummary
                ))
            .call()
            .content();
        
        return parseDedupResult(dedupJson);
    }
    
    /**
     * 批量事件识别（更高层次）
     * 
     * 同一问题短时间内被多人投诉 → 可能是群体性事件
     */
    public BatchEventDetect detectBatchEvent(String district, 
                                              String category) {
        
        // 过去2小时同区域同分类的工单数
        long count = orderRepository.countByDistrictAndCategoryInTimeRange(
            district, category, 
            LocalDateTime.now().minusHours(2), 
            LocalDateTime.now()
        );
        
        if (count >= 5) {
            return BatchEventDetect.builder()
                .isBatchEvent(true)
                .district(district)
                .category(category)
                .orderCount(count)
                .alertLevel(count >= 10 ? "红色预警" : "橙色预警")
                .suggestion(String.format(
                    "%s区域%s类诉求2小时内已达%d件，"
                    + "建议启动群体性事件应急预案",
                    district, category, count
                ))
                .build();
        }
        
        return BatchEventDetect.notBatch();
    }
}
```

### 3.3 标准答复话术生成

```java
@Service
public class ReplyScriptGenerator {
    
    private final ChatClient chatClient;
    private final KnowledgeBaseService kbService;
    
    /**
     * 实时生成标准答复话术
     * 
     * 针对不同类型诉求生成不同风格的话术
     */
    public ReplyScript generateScript(String transcript,
                                       ClassificationResult classification,
                                       String city) {
        
        switch (classification.getLevel1()) {
            case "咨询" -> {
                return generateConsultationScript(transcript, classification, city);
            }
            case "投诉" -> {
                return generateComplaintScript(transcript, classification, city);
            }
            case "举报" -> {
                return generateReportScript(transcript, classification, city);
            }
            case "建议" -> {
                return generateSuggestionScript(transcript, classification, city);
            }
            default -> {
                return generateGeneralScript(transcript, classification, city);
            }
        }
    }
    
    /**
     * 咨询类话术
     */
    private ReplyScript generateConsultationScript(String transcript,
                                                     ClassificationResult cls,
                                                     String city) {
        
        // 1. 从知识库检索相关政策和问答
        List<KnowledgeArticle> articles = kbService.search(
            transcript, cls, city, 3
        );
        
        // 2. 生成话术
        String script = chatClient.prompt()
            .system("""
                你是%s12345热线的标准话术生成专家。
                
                话术要求：
                1. 开头：礼貌问候 + 确认理解对方问题
                2. 中间：引用具体政策条款 + 清晰解释
                3. 结尾：确认对方是否理解 + 是否需要进一步帮助
                4. 口语化但保持专业（这是电话沟通，不是写公文）
                5. 如果涉及具体办事流程，告诉对方"去哪里、带什么、找谁"
                
                不要说的：
                - "根据相关规定"（太模糊）
                - "好像是"（不确定）
                - "你去找XX部门"（踢皮球）
                """.formatted(city))
            .user("""
                市民问题：%s
                分类：%s
                
                相关知识库资料：
                %s
                
                请生成标准话术（200-400字）。
                """.formatted(
                    transcript,
                    cls.toPathString(),
                    articles.stream()
                        .map(a -> String.format("【%s】\n%s", a.getTitle(), a.getContent()))
                        .collect(Collectors.joining("\n\n"))
                ))
            .call()
            .content();
        
        return ReplyScript.builder()
            .type(ScriptType.CONSULTATION)
            .content(script)
            .references(articles.stream()
                .map(KnowledgeArticle::getSource)
                .collect(Collectors.toList()))
            .keyPoints(extractKeyPoints(script))
            .build();
    }
    
    /**
     * 投诉类话术
     */
    private ReplyScript generateComplaintScript(String transcript,
                                                 ClassificationResult cls,
                                                 String city) {
        
        String script = chatClient.prompt()
            .system("""
                你是%s12345热线的投诉受理专家。
                
                投诉受理话术架构（五步法）：
                1. 共情："我理解您的心情..."
                2. 确认："您是说XX地方的XX问题，对吗？"
                3. 告知："这个情况我会记录并转给XX部门处理"
                4. 承诺时限："按规定XX个工作日内会向您反馈结果"
                5. 闭环："您的工单号是XX，也可以通过XX渠道查询进度"
                
                原则：
                - 不承诺做不到的事（如"马上解决"）
                - 给出明确的受理编号
                - 告知后续查询方式
                """.formatted(city))
            .user("""
                市民投诉：%s
                分类：%s → %s → %s
                责任部门：%s
                法定处理时限：%d个工作日
                
                请生成受理话术。
                """.formatted(
                    transcript,
                    cls.getLevel1(), cls.getLevel2(), cls.getLevel3(),
                    cls.getResponsibleDepartment(),
                    cls.getLegalProcessingDays()
                ))
            .call()
            .content();
        
        return ReplyScript.builder()
            .type(ScriptType.COMPLAINT)
            .content(script)
            .build();
    }
}
```

```java
@Service
public class HotspotAnalysisService {
    
    /**
     * 每日热点分析报告
     */
    @Scheduled(cron = "0 0 7 * * ?")  // 每天早上7点
    public void generateDailyHotspotReport() {
        
        LocalDate yesterday = LocalDate.now().minusDays(1);
        
        // 1. 统计昨日各类诉求的数量
        List<CategoryStats> stats = orderRepository
            .countByCategoryAndDate(yesterday);
        
        // 2. 对比前7日平均值，找异常增长的分类
        List<HotspotAlert> hotspots = findHotspots(stats); 
        
        // 3. 分析高频关键词
        Map<String, Long> keywordTrends = analyzeKeywordTrends(yesterday);
        
        // 4. 生成区域热力图数据
        Map<String, Integer> districtHeatmap = generateDistrictHeatmap(yesterday);
        
        // 5. LLM生成综合分析报告
        String report = generateReport(stats, hotspots, keywordTrends, 
                                        districtHeatmap, yesterday);
        
        // 6. 推送给相关领导/部门
        notificationService.sendToLeadership(report, hotspots);
    }
    
    private List<HotspotAlert> findHotspots(List<CategoryStats> todayStats) {
        
        List<HotspotAlert> hotspots = new ArrayList<>();
        
        for (CategoryStats stat : todayStats) {
            double avgLast7Days = stat.getAvgLast7Days();
            
            if (avgLast7Days > 0) {
                double increaseRate = (stat.getCount() - avgLast7Days) / avgLast7Days;
                
                if (increaseRate > 1.0 && stat.getCount() > 20) {
                    // 翻倍增长且绝对数量>20 → 热点
                    hotspots.add(HotspotAlert.builder()
                        .category(stat.getCategory())
                        .todayCount(stat.getCount())
                        .avgLast7Days(avgLast7Days)
                        .increaseRate(increaseRate)
                        .level(increaseRate > 2.0 ? 
                            AlertLevel.CRITICAL : AlertLevel.HIGH)
                        .build());
                } else if (increaseRate > 0.5 && stat.getCount() > 50) {
                    // 增长50%且绝对数量>50 → 关注
                    hotspots.add(HotspotAlert.builder()
                        .category(stat.getCategory())
                        .todayCount(stat.getCount())
                        .avgLast7Days(avgLast7Days)
                        .increaseRate(increaseRate)
                        .level(AlertLevel.WARNING)
                        .build());
                }
            }
        }
        
        hotspots.sort(Comparator.comparingDouble(
            HotspotAlert::getIncreaseRate).reversed());
        
        return hotspots;
    }
    
    private String generateReport(List<CategoryStats> stats,
                                   List<HotspotAlert> hotspots,
                                   Map<String, Long> keywordTrends,
                                   Map<String, Integer> districtHeatmap,
                                   LocalDate date) {
        
        return chatClient.prompt()
            .system("""
                你是政府社情民意分析专家。撰写12345热线日度分析报告。
                
                报告结构：
                1. 总体概况（昨日工单总量、与昨日/上周对比）
                2. 热点聚焦（TOP3异常增长的诉求类别及原因分析）
                3. 区域分布（各区县诉求密度及特点）
                4. 高频关键词（市民关注焦点的词云分析）
                5. 风险预警（需要引起重视的苗头性问题）
                6. 处置建议（给相关部门的工作建议）
                
                风格：数据驱动的简明报告，每部分2-3句话。
                """)
            .user("""
                %s市12345热线 %s 日度分析：
                
                总量：%d件（较前日%s%.0f%%，较上周同期%s%.0f%%）
                
                异常增长分类（TOP5）：
                %s
                
                各区诉求分布：
                %s
                
                高频关键词：
                %s
                """.formatted(
                    stats.get(0).getCity(),
                    date,
                    stats.stream().mapToLong(CategoryStats::getCount).sum(),
                    "增长", 1.0,
                    "增长", 2.0,
                    hotspots.stream()
                        .map(h -> String.format("%s：%d件（增长%.0f%%）",
                            h.getCategory(), h.getTodayCount(), 
                            h.getIncreaseRate() * 100))
                        .collect(Collectors.joining("\n")),
                    districtHeatmap.entrySet().stream()
                        .sorted(Map.Entry.<String, Integer>comparingByValue().reversed())
                        .map(e -> e.getKey() + "：" + e.getValue() + "件")
                        .collect(Collectors.joining("；")),
                    keywordTrends.entrySet().stream()
                        .sorted(Map.Entry.<String, Long>comparingByValue().reversed())
                        .limit(20)
                        .map(e -> e.getKey() + "(" + e.getValue() + "次)")
                        .collect(Collectors.joining("、"))
                ))
            .call()
            .content();
    }
}
```
