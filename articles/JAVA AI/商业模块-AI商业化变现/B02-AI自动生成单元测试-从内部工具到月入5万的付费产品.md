### 4.1 Java源码AST分析

```java
/**
 * Java源码分析器
 * 用AST提取方法签名、参数类型、依赖注入等信息
 */
@Component
public class JavaCodeAnalyzer {
    
    /**
     * 解析Java源文件，提取测试生成所需的所有元信息
     */
    public ClassMetaInfo analyze(String sourceCode) {
        ClassMetaInfo meta = new ClassMetaInfo();
        
        // 使用JavaParser解析源码
        CompilationUnit cu = StaticJavaParser.parse(sourceCode);
        
        // 1. 提取类信息
        cu.findAll(ClassOrInterfaceDeclaration.class).forEach(clazz -> {
            meta.setClassName(clazz.getNameAsString());
            meta.setPackageName(cu.getPackageDeclaration()
                .map(pd -> pd.getNameAsString()).orElse(""));
            meta.setIsInterface(clazz.isInterface());
            meta.setIsAbstract(clazz.isAbstract());
            
            // 提取继承和实现
            clazz.getExtendedTypes().forEach(t -> 
                meta.getSuperClass(t.getNameAsString()));
            clazz.getImplementedTypes().forEach(t -> 
                meta.getInterfaces().add(t.getNameAsString()));
        });
        
        // 2. 提取字段信息（含依赖注入注解）
        parseFields(cu, meta);
        
        // 3. 提取所有公共方法信息
        parseMethods(cu, meta);
        
        // 4. 提取构造函数信息
        parseConstructors(cu, meta);
        
        // 5. 分析Spring注解（@Service/@Component/@Repository等）
        meta.setSpringAnnotations(
            cu.findAll(AnnotationExpr.class).stream()
                .filter(a -> a.getNameAsString().startsWith("Spring") 
                    || a.getNameAsString().equals("Service")
                    || a.getNameAsString().equals("Component")
                    || a.getNameAsString().equals("Repository")
                    || a.getNameAsString().equals("RestController"))
                .map(AnnotationExpr::getNameAsString)
                .collect(Collectors.toSet())
        );
        
        return meta;
    }
    
    /**
     * 解析类字段，识别依赖注入
     */
    private void parseFields(CompilationUnit cu, ClassMetaInfo meta) {
        cu.findAll(FieldDeclaration.class).forEach(field -> {
            FieldDeclaration fd = (FieldDeclaration) field;
            
            fd.getVariables().forEach(var -> {
                FieldInfo fi = FieldInfo.builder()
                    .name(var.getNameAsString())
                    .type(fd.getElementType().asString())
                    .build();
                
                // 检查注解：@Autowired, @Mock, @InjectMocks
                fd.getAnnotations().forEach(ann -> {
                    String annName = ann.getNameAsString();
                    if (annName.equals("Autowired") || annName.equals("Inject")) {
                        fi.setAutowired(true);
                        meta.getDependencies().add(fi);
                    }
                    if (annName.equals("Value")) {
                        fi.setConfigValue(true);
                    }
                });
                
                meta.getAllFields().add(fi);
            });
        });
    }
    
    /**
     * 解析方法，提取参数、返回类型、异常声明等
     * 这些信息是生成测试用例的关键
     */
    private void parseMethods(CompilationUnit cu, ClassMetaInfo meta) {
        cu.findAll(MethodDeclaration.class).stream()
            .filter(m -> m.isPublic() && !m.isStatic())
            .forEach(method -> {
                MethodInfo mi = MethodInfo.builder()
                    .name(method.getNameAsString())
                    .returnType(method.getTypeAsString())
                    .isVoid(method.getType().isVoidType())
                    .parameters(parseParameters(method))
                    .thrownExceptions(method.getThrownExceptions().stream()
                        .map(t -> t.getNameAsString())
                        .collect(Collectors.toList()))
                    .body(method.getBody().map(Node::toString).orElse(""))
                    .isTransactional(
                        method.getAnnotationByName("Transactional").isPresent()
                    )
                    .build();
                
                // 分析方法复杂度
                mi.setCyclomaticComplexity(calculateComplexity(method));
                
                // 提取方法中的if/switch分支（用于生成边界测试）
                mi.setConditionalBranches(extractConditionalBranches(method));
                
                meta.getPublicMethods().add(mi);
            });
    }
    
    /**
     * 计算圈复杂度（用于判断测试优先级）
     */
    private int calculateComplexity(MethodDeclaration method) {
        int complexity = 1; // 基础复杂度
        String body = method.getBody().map(Node::toString).orElse("");
        
        complexity += countOccurrences(body, "if (");
        complexity += countOccurrences(body, "else if");
        complexity += countOccurrences(body, "case ");
        complexity += countOccurrences(body, "for (");
        complexity += countOccurrences(body, "while (");
        complexity += countOccurrences(body, "&&");
        complexity += countOccurrences(body, "||");
        complexity += countOccurrences(body, "catch (");
        complexity += countOccurrences(body, "?:");
        
        return complexity;
    }
    
    /**
     * 提取方法中的条件分支（用于生成针对性测试）
     */
    private List<ConditionalBranch> extractConditionalBranches(MethodDeclaration method) {
        List<ConditionalBranch> branches = new ArrayList<>();
        
        method.getBody().ifPresent(body -> {
            // 找到所有if语句
            body.findAll(IfStmt.class).forEach(ifStmt -> {
                branches.add(ConditionalBranch.builder()
                    .type("IF")
                    .condition(ifStmt.getCondition().toString())
                    .lineNumber(ifStmt.getBegin().map(p -> p.line).orElse(-1))
                    .build());
            });
            
            // 找到所有switch语句
            body.findAll(SwitchStmt.class).forEach(switchStmt -> {
                switchStmt.getEntries().forEach(entry -> {
                    entry.getLabels().forEach(label -> {
                        branches.add(ConditionalBranch.builder()
                            .type("SWITCH_CASE")
                            .condition(label.toString())
                            .lineNumber(entry.getBegin().map(p -> p.line).orElse(-1))
                            .build());
                    });
                });
            });
        });
        
        return branches;
    }
}

### 4.2 AI测试生成引擎

```java
/**
 * AI测试生成引擎
 * 将Java代码分析结果 + Prompt → 生成JUnit 5测试代码
 */
@Service
@Slf4j
public class TestGenerationEngine {
    
    @Autowired
    private ChatLanguageModel chatModel;
    
    @Autowired
    private JavaCodeAnalyzer codeAnalyzer;
    
    /**
     * 主入口：为一个Java类生成完整测试
     */
    public GeneratedTest generateTest(String sourceCode, TestGenOptions options) {
        
        // Step 1: 分析源码
        ClassMetaInfo meta = codeAnalyzer.analyze(sourceCode);
        
        // Step 2: 为每个公共方法生成测试
        List<TestMethod> testMethods = new ArrayList<>();
        
        for (MethodInfo method : meta.getPublicMethods()) {
            // 跳过getter/setter/toString等简单方法（可选生成）
            if (options.isSkipSimpleMethods() && isSimpleMethod(method)) {
                continue;
            }
            
            // 生成基础功能测试
            testMethods.addAll(generateMethodTests(method, meta, options));
            
            // 生成边界条件测试
            if (options.isGenerateBoundaryTests()) {
                testMethods.addAll(generateBoundaryTests(method, meta, options));
            }
            
            // 生成异常场景测试
            testMethods.addAll(generateExceptionTests(method, meta, options));
        }
        
        // Step 3: 组装完整的测试类
        String testClassName = meta.getClassName() + "Test";
        String packageName = meta.getPackageName();
        
        String testClassCode = assembleTestClass(
            testClassName, packageName, meta, testMethods, options
        );
        
        return GeneratedTest.builder()
            .className(testClassName)
            .packageName(packageName)
            .sourceCode(testClassCode)
            .testMethods(testMethods)
            .estimatedCoverage(estimateCoverage(meta, testMethods))
            .build();
    }
    
    /**
     * 为一个方法生成基础功能测试
     */
    private List<TestMethod> generateMethodTests(
            MethodInfo method, ClassMetaInfo meta, TestGenOptions options) {
        
        String prompt = buildMethodTestPrompt(method, meta, options);
        String aiResponse = chatModel.generate(prompt);
        
        return parseTestMethodResponse(aiResponse, method.getName());
    }
    
    /**
     * 构建方法测试Prompt
     */
    private String buildMethodTestPrompt(
            MethodInfo method, ClassMetaInfo meta, TestGenOptions options) {
        
        return """
            # 角色
            你是一个资深的Java单元测试专家，精通JUnit 5 + Mockito。
            
            # 任务
            为以下Java方法生成完整的单元测试。
            
            # 被测类信息
            类名：%s
            包名：%s
            Spring注解：%s
            
            # 被测方法
            ```java
            %s %s(%s) %s {
                %s
            }
            ```
            
            # 依赖注入字段（需要Mock）
            %s
            
            # 测试框架要求
            - 使用 JUnit 5 (Jupiter)
            - 使用 Mockito 进行Mock
            - 使用 @ExtendWith(MockitoExtension.class)
            - 使用 @Mock 标注需要Mock的依赖
            - 使用 @InjectMocks 标注被测类
            
            # 测试覆盖要求
            1. 正常场景（Happy Path）
            2. %s
            
            # 方法中的条件分支（请确保每个分支都有测试覆盖）
            %s
            
            # 命名规范
            - 测试方法命名：test[方法名]_[场景描述]_[预期结果]
            - 示例：testCalculateDiscount_WithVIPUser_ShouldApplyVIPRate
            
            # 代码风格
            - 使用 AssertJ (assertThat) 断言
            - 使用 BDDMockito (given/willReturn)
            - 测试方法不超过30行
            - 适当添加 @DisplayName 中文描述
            
            # 重要提醒
            - 必须Mock所有外部依赖（数据库、HTTP、文件IO等）
            - 不要假设数据存在，测试数据自己构造
            - 所有测试方法必须在10秒内完成（不要有Thread.sleep）
            
            # 输出格式
            只输出Java代码，不要解释。代码可以直接编译运行。
            每个测试方法用 @Test 标注。
            """.formatted(
                meta.getClassName(),
                meta.getPackageName(),
                meta.getSpringAnnotations(),
                method.getReturnType(),
                method.getName(),
                formatParameters(method.getParameters()),
                formatThrows(method.getThrownExceptions()),
                formatMethodBodySummary(method.getBody()),
                formatDependencies(meta.getDependencies()),
                options.getTestStrategy().getDescription(),
                formatBranches(method.getConditionalBranches())
            );
    }
    
    /**
     * 生成边界条件测试
     */
    private List<TestMethod> generateBoundaryTests(
            MethodInfo method, ClassMetaInfo meta, TestGenOptions options) {
        
        List<TestMethod> boundaryTests = new ArrayList<>();
        
        // 1. 参数为null的测试
        method.getParameters().stream()
            .filter(p -> !p.isPrimitive())
            .forEach(param -> {
                String prompt = """
                    为方法 %s.%s 生成一个参数 `%s` 为 null 的边界测试。
                    测试应该验证方法正确处理null输入（抛出异常或返回安全默认值）。
                    
                    方法签名：%s
                    
                    只输出测试方法代码，用 @Test 标注。
                    """.formatted(
                        meta.getClassName(), method.getName(),
                        param.getName(),
                        method.getSignature()
                    );
                
                String response = chatModel.generate(prompt);
                boundaryTests.addAll(parseTestMethodResponse(response, 
                    method.getName() + "_NullParam"));
            });
        
        // 2. 空集合/空字符串的测试
        method.getParameters().stream()
            .filter(p -> p.getType().equals("String") 
                || p.getType().startsWith("List") 
                || p.getType().startsWith("Set") 
                || p.getType().startsWith("Map"))
            .forEach(param -> {
                String prompt = """
                    为方法 %s.%s 生成参数 `%s` 为空%s的边界测试。
                    测试应验证方法正确处理空值输入。
                    
                    只输出测试方法代码，用 @Test 标注。
                    """.formatted(
                        meta.getClassName(), method.getName(),
                        param.getName(),
                        p.getType().startsWith("String") ? "字符串\"\"" : "集合"
                    );
                
                String response = chatModel.generate(prompt);
                boundaryTests.addAll(parseTestMethodResponse(response, 
                    method.getName() + "_EmptyValue"));
            });
        
        return boundaryTests;
    }
    
    /**
     * 生成异常场景测试
     */
    private List<TestMethod> generateExceptionTests(
            MethodInfo method, ClassMetaInfo meta, TestGenOptions options) {
        
        List<TestMethod> exceptionTests = new ArrayList<>();
        
        // 1. 方法声明了throws → 生成异常被抛出的测试
        for (String exceptionType : method.getThrownExceptions()) {
            String prompt = """
                为方法 %s.%s 生成一个测试：验证当依赖的方法抛出 %s 时，被测试方法的行为。
                
                使用 Mockito 的 given().willThrow() 来模拟异常。
                如果方法应该抛出异常，使用 assertThrows()。
                如果方法应该处理异常（catch后不抛出），验证返回值和日志。
                
                只输出测试方法代码，用 @Test 标注。
                """.formatted(
                    meta.getClassName(), method.getName(), exceptionType
                );
            
            String response = chatModel.generate(prompt);
            exceptionTests.addAll(parseTestMethodResponse(response, 
                method.getName() + "_Throws_" + exceptionType));
        }
        
        // 2. 数据库/网络异常场景
        if (method.isCallsDatabase() || method.isCallsExternal()) {
            String prompt = """
                为方法 %s.%s 生成一个外部依赖失败的测试。
                模拟场景：%s
                验证方法有适当的错误处理和回退机制。
                
                只输出测试方法代码，用 @Test 标注。
                """.formatted(
                    meta.getClassName(), method.getName(),
                    method.isCallsDatabase() ? "数据库连接超时" : "外部API调用失败"
                );
            
            String response = chatModel.generate(prompt);
            exceptionTests.addAll(parseTestMethodResponse(response, 
                method.getName() + "_ExternalFailure"));
        }
        
        return exceptionTests;
    }
}

### 4.3 智能Mock生成

```java
/**
 * AI驱动的Mock配置生成
 * 自动分析依赖关系，生成正确的Mockito配置
 */
@Component
public class MockConfigGenerator {
    
    /**
     * 分析被测类的所有依赖，生成Mock配置
     */
    public MockConfiguration generateMockConfig(ClassMetaInfo meta, 
                                                  List<MethodInfo> methods) {
        
        MockConfiguration config = new MockConfiguration();
        
        // 1. 为每个依赖生成Mock声明
        for (FieldInfo dependency : meta.getDependencies()) {
            MockDeclaration mock = MockDeclaration.builder()
                .fieldName(dependency.getName())
                .fieldType(dependency.getType())
                .annotation("@Mock")
                .build();
            config.getMocks().add(mock);
        }
        
        // 2. 分析每个方法需要的Mock行为
        for (MethodInfo method : methods) {
            // 提取方法体中调用的依赖方法
            DependencyCalls dependencyCalls = analyzeDependencyCalls(
                method, meta.getDependencies()
            );
            
            // 为每个依赖调用生成Mock行为配置
            for (DependencyCall call : dependencyCalls.getCalls()) {
                MockBehavior behavior = generateMockBehavior(call, method);
                config.getMockBehaviors().add(behavior);
            }
        }
        
        // 3. 处理@Value注解的字段
        for (FieldInfo field : meta.getAllFields()) {
            if (field.isConfigValue()) {
                config.getReflectionTestUtilsConfigs().add(
                    new ReflectionConfig(field.getName(), generateTestValue(field))
                );
            }
        }
        
        return config;
    }
    
    /**
     * 智能生成Mock方法的返回值
     */
    private MockBehavior generateMockBehavior(DependencyCall call, MethodInfo context) {
        
        String returnType = call.getReturnType();
        
        // 根据返回类型生成合适的模拟对象
        Object mockReturn = switch (returnType) {
            case "String" -> "\"mock-" + call.getMethodName() + "-result\"";
            case "int", "Integer" -> 42;
            case "long", "Long" -> 42L;
            case "boolean", "Boolean" -> true;
            case "List" -> "List.of()";
            case "Optional" -> "Optional.empty()";
            case "void" -> null; // void方法不需要返回值
            default -> {
                // 复杂对象，生成Builder模式构造
                yield generateComplexMockObject(returnType);
            }
        };
        
        return MockBehavior.builder()
            .mockField(call.getMockFieldName())
            .methodName(call.getMethodName())
            .methodParameters(call.getParameters())
            .returnValue(mockReturn)
            .build();
    }
    
    /**
     * 为复杂对象生成构造代码
     */
    private String generateComplexMockObject(String typeName) {
        String prompt = """
            为一个Java类型 %s 生成一个构造示例。
            使用Builder模式或构造函数，创建该对象的一个合理实例。
            
            要求：
            - 字段值要有真实感（不是 "test" "123" 这种）
            - 如果类型有Builder，使用Builder
            - 只输出构造代码，不要类定义
            
            示例输出风格：
            User.builder()
                .id(1L)
                .name("张三")
                .email("zhangsan@example.com")
                .build()
            """.formatted(typeName);
        
        return chatModel.generate(prompt);
    }
}
