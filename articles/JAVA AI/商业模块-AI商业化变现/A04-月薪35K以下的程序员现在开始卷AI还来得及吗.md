```java
// 第2周就能写出来的代码
@RestController
public class SimpleAiController {
    
    @Value("${ai.api.key}")
    private String apiKey;
    
    @Value("${ai.api.url}")
    private String apiUrl;
    
    private final RestTemplate restTemplate = new RestTemplate();
    
    @PostMapping("/chat")
    public ResponseEntity<String> chat(@RequestBody ChatRequest chatRequest) {
        
        // 构建System Prompt + User Prompt
        String systemPrompt = """
            你是一个经验丰富的Java技术面试官。
            你的回答应该：
            1. 包含技术原理分析
            2. 给出实际应用场景
            3. 指出常见的理解误区
            请用中文回答。
            """;
        
        Map<String, Object> requestBody = new HashMap<>();
        requestBody.put("model", "gpt-4");
        requestBody.put("messages", List.of(
            Map.of("role", "system", "content", systemPrompt),
            Map.of("role", "user", "content", chatRequest.getMessage())
        ));
        requestBody.put("temperature", 0.7);
        requestBody.put("max_tokens", 2000);
        
        HttpHeaders headers = new HttpHeaders();
        headers.set("Authorization", "Bearer " + apiKey);
        headers.setContentType(MediaType.APPLICATION_JSON);
        
        HttpEntity<Map<String, Object>> entity = new HttpEntity<>(requestBody, headers);
        
        ResponseEntity<Map> response = restTemplate.postForEntity(
            apiUrl, entity, Map.class
        );
        
        Map<String, Object> body = response.getBody();
        List<Map<String, Object>> choices = (List<Map<String, Object>>) body.get("choices");
        Map<String, Object> message = (Map<String, Object>) choices.get(0).get("message");
        String content = (String) message.get("content");
        
        return ResponseEntity.ok(content);
    }
}
```

```java
@Service
public class DocumentProcessor {
    
    @Autowired
    private EmbeddingClient embeddingClient;
    
    @Autowired
    private VectorStore vectorStore;
    
    /**
     * 处理用户上传的知识文档
     */
    public int processDocument(MultipartFile file, String tenantId) {
        int chunkCount = 0;
        
        try {
            // 1. 文档解析（支持PDF、Word、TXT等）
            String rawText = parseDocument(file);
            
            // 2. 自适应文档切分
            List<String> chunks = splitDocument(rawText, 
                SplitConfig.builder()
                    .maxChunkSize(500)      // 每段最多500字
                    .overlapSize(100)       // 段间重叠100字
                    .preserveParagraphs(true) // 保留段落完整性
                    .build()
            );
            
            // 3. 批量向量化并存储
            List<String> chunkTexts = new ArrayList<>();
            for (int i = 0; i < chunks.size(); i++) {
                String chunk = chunks.get(i);
                
                float[] embedding = embeddingClient.embed(chunk);
                
                Document doc = Document.builder()
                    .content(chunk)
                    .embedding(embedding)
                    .metadata(Map.of(
                        "tenant_id", tenantId,
                        "file_name", file.getOriginalFilename(),
                        "chunk_index", i,
                        "section", detectSection(chunk),
                        "created_at", Instant.now().toString()
                    ))
                    .build();
                
                vectorStore.add(doc);
                chunkCount++;
            }
            
        } catch (Exception e) {
            log.error("文档处理失败: {}", file.getOriginalFilename(), e);
            throw new DocumentProcessException("文档处理失败", e);
        }
        
        return chunkCount;
    }
    
    private List<String> splitDocument(String text, SplitConfig config) {
        // 按段落优先切分，避免在句子中间断开
        String[] paragraphs = text.split("\n\\s*\n");
        List<String> chunks = new ArrayList<>();
        StringBuilder currentChunk = new StringBuilder();
        
        for (String paragraph : paragraphs) {
            if (currentChunk.length() + paragraph.length() > config.getMaxChunkSize()
                && currentChunk.length() > 0) {
                chunks.add(currentChunk.toString().trim());
                // 保留重叠部分
                String overlap = currentChunk.substring(
                    Math.max(0, currentChunk.length() - config.getOverlapSize())
                );
                currentChunk = new StringBuilder(overlap);
            }
            currentChunk.append(paragraph).append("\n\n");
        }
        
        if (currentChunk.length() > 0) {
            chunks.add(currentChunk.toString().trim());
        }
        
        return chunks;
    }
}
```

**第6周：向量检索与生成**

```java
@Service
public class RAGService {
    
    @Autowired
    private VectorStore vectorStore;
    
    @Autowired
    private EmbeddingClient embeddingClient;
    
    @Autowired
    private ChatClient chatClient;
    
    /**
     * RAG完整流程：检索 → 增强 → 生成
     */
    public RAGResponse generateAnswer(String question, String tenantId) {
        
        // Step 1: 检索相关文档
        List<Document> relevantDocs = retrieveRelevantDocuments(question, tenantId);
        
        if (relevantDocs.isEmpty()) {
            return RAGResponse.knowledgeNotFound("抱歉，知识库中没有找到相关信息。");
        }
        
        // Step 2: 重排序（Rerank），把最相关的排前面
        relevantDocs = rerankDocuments(question, relevantDocs);
        
        // Step 3: 构建增强Prompt
        String augmentedPrompt = buildAugmentedPrompt(question, relevantDocs);
        
        // Step 4: 生成回答
        String answer = chatClient.call(augmentedPrompt);
        
        // Step 5: 引用标注
        List<Citation> citations = extractCitations(answer, relevantDocs);
        
        return RAGResponse.builder()
            .answer(answer)
            .citations(citations)
            .retrievedDocCount(relevantDocs.size())
            .build();
    }
    
    private List<Document> retrieveRelevantDocuments(String question, String tenantId) {
        // 问题向量化
        float[] questionEmbedding = embeddingClient.embed(question);
        
        // 向量检索
        SearchResult result = vectorStore.search(
            SearchRequest.builder()
                .vector(questionEmbedding)
                .topK(5)
                .threshold(0.7f)
                .filter(Map.of("tenant_id", tenantId))
                .build()
        );
        
        return result.getDocuments();
    }
    
    private String buildAugmentedPrompt(String question, List<Document> docs) {
        StringBuilder sb = new StringBuilder();
        sb.append("""
            你是企业的AI知识助手。请基于以下参考文档回答用户问题。
            
            要求：
            1. 如果参考文档包含答案，请准确引用，并标注【参考：文档X】
            2. 如果参考文档不完整，可以结合你的知识补充，但要明确说明
            3. 如果参考文档完全不能回答问题，请如实告知
            
            """);
        
        sb.append("=== 参考文档 ===\n");
        for (int i = 0; i < docs.size(); i++) {
            sb.append("【文档").append(i + 1).append("】\n");
            sb.append(docs.get(i).getContent()).append("\n\n");
        }
        
        sb.append("=== 用户问题 ===\n");
        sb.append(question).append("\n\n");
        sb.append("请回答：");
        
        return sb.toString();
    }
}
```

```java
// Self-RAG：AI自我反思式检索
@Service
public class SelfRAGService {
    
    public String selfRAGQuery(String question, String tenantId) {
        // 第1轮：第一次检索和生成
        List<Document> docs1 = vectorStore.search(question, tenantId, 3);
        String answer1 = generateWithDocuments(question, docs1);
        
        // 第2轮：AI自我检查——是否需要补充检索？
        String reflection = chatClient.call("""
            请评估以下回答是否充分：
            问题：%s
            回答：%s
            
            如果回答足够好，回复"OK"。
            如果需要补充信息，回复"MORE: <需要补充的关键词>"。
            """.formatted(question, answer1));
        
        if (reflection.startsWith("MORE:")) {
            String supplementKeyword = reflection.substring(5).trim();
            
            // 第3轮：用补充关键词再次检索
            List<Document> docs2 = vectorStore.search(supplementKeyword, tenantId, 3);
            
            // 合并去重
            Set<String> seenContents = new HashSet<>();
            List<Document> mergedDocs = new ArrayList<>();
            for (Document doc : docs1) {
                if (seenContents.add(doc.getContent())) {
                    mergedDocs.add(doc);
                }
            }
            for (Document doc : docs2) {
                if (seenContents.add(doc.getContent())) {
                    mergedDocs.add(doc);
                }
            }
            
            return generateWithDocuments(question, mergedDocs);
        }
        
        return answer1;
    }
}
```

```java
// 多Agent协作示例
@Service
public class MultiAgentOrchestrator {
    
    @Autowired
    private DevOpsAgent devOpsAgent;
    
    @Autowired
    private CodeReviewAgent codeReviewAgent;
    
    @Autowired
    private SecurityAuditAgent securityAgent;
    
    /**
     * 多Agent流水线
     */
    public AuditReport comprehensiveAudit(String projectName) {
        
        AuditReport report = new AuditReport();
        report.setProjectName(projectName);
        
        // Step 1: 代码审查Agent
        CompletableFuture<CodeReviewResult> codeReviewFuture = 
            CompletableFuture.supplyAsync(() -> 
                codeReviewAgent.review(projectName)
            );
        
        // Step 2: 安全检查Agent
        CompletableFuture<SecurityResult> securityFuture = 
            CompletableFuture.supplyAsync(() -> 
                securityAgent.audit(projectName)
            );
        
        // Step 3: 等待两个Agent完成
        CompletableFuture.allOf(codeReviewFuture, securityFuture).join();
        
        report.setCodeReview(codeReviewFuture.join());
        report.setSecurityAudit(securityFuture.join());
        
        // Step 4: DevOps Agent汇总
        String summary = devOpsAgent.summarizeReport(report);
        report.setSummary(summary);
        
        return report;
    }
}
```

## 四、常见顾虑解答

### 顾虑一："我已经35岁了，学得动吗？"

我团队里有个40岁的同事，去年开始学AI，半年后成了公司的AI平台负责人。编程经验是你的优势，不是劣势。AI工程化需要的是工程经验，不是年轻的身体。

### 顾虑二："我学历不高，竞争不过名校生"

企业要找的是能干活的人，不是学历。我见过太多985毕业的Python程序员，RAG做得一塌糊涂，因为他们不懂企业级工程。你的Java工程经验是稀缺资产。

### 顾虑三："AI会不会只是昙花一现？"

如果2023年初你问这个问题，我理解。但现在是2026年，GPT从3.5进化到4.5，Claude到4.0，DeepSeek震惊全球。AI已经渗透到每个行业。这不是泡沫，是生产力革命。

### 顾虑四："我学AI要辞职全职学吗？"
