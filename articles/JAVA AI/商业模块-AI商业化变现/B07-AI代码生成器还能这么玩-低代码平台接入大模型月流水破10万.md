### 3.2 模板引擎——代码生成的基石

```java
/**
 * 模板驱动的代码生成器
 * 使用Freemarker + AI填充业务逻辑
 */
@Service
public class CodeGenerationEngine {
    
    private final Configuration freemarkerConfig;
    
    public CodeGenerationEngine() {
        freemarkerConfig = new Configuration(Configuration.VERSION_2_3_32);
        freemarkerConfig.setClassForTemplateLoading(this.getClass(), "/templates");
        freemarkerConfig.setDefaultEncoding("UTF-8");
    }
    
    /**
     * 生成完整的业务模块代码
     */
    public GeneratedModule generateModule(TechnicalSpec spec, CodeGenOptions options) {
        
        GeneratedModule module = new GeneratedModule();
        
        for (EntitySpec entity : spec.getEntities()) {
            
            // 准备模板数据
            Map<String, Object> templateData = buildTemplateData(spec, entity, options);
            
            // 生成各个层次的代码
            module.addFile(
                entity.getName() + ".java",
                "entity", 
                renderTemplate("entity.ftl", templateData)
            );
            
            module.addFile(
                entity.getName() + "Mapper.java",
                "mapper",
                renderTemplate("mapper.ftl", templateData)
            );
            
            module.addFile(
                entity.getName() + "Mapper.xml",
                "mapper/xml",
                renderTemplate("mapper_xml.ftl", templateData)
            );
            
            module.addFile(
                "I" + entity.getName() + "Service.java",
                "service",
                renderTemplate("service_interface.ftl", templateData)
            );
            
            module.addFile(
                entity.getName() + "ServiceImpl.java",
                "service/impl",
                renderTemplate("service_impl.ftl", templateData)
            );
            
            module.addFile(
                entity.getName() + "Controller.java",
                "controller",
                renderTemplate("controller.ftl", templateData)
            );
            
            // 生成DTO
            module.addFile(
                entity.getName() + "CreateDTO.java",
                "dto",
                renderTemplate("create_dto.ftl", templateData)
            );
            
            module.addFile(
                entity.getName() + "QueryDTO.java",
                "dto", 
                renderTemplate("query_dto.ftl", templateData)
            );
            
            module.addFile(
                entity.getName() + "VO.java",
                "vo",
                renderTemplate("vo.ftl", templateData)
            );
            
            // AI生成复杂业务逻辑（Service中模板无法覆盖的部分）
            if (!entity.getBusinessRules().isEmpty()) {
                String businessLogic = generateBusinessLogic(entity, spec);
                mergeBusinessLogic(module, entity.getName(), businessLogic);
            }
        }
        
        // 生成建表DDL
        module.setDdl(generateDDL(spec));
        
        // 生成前端代码
        module.setFrontendCode(generateFrontend(spec, options));
        
        // 生成配置文件
        if (options.isAutoConfig()) {
            module.setConfigFiles(generateConfigFiles(spec));
        }
        
        return module;
    }
    
    /**
     * Freemarker模板渲染
     */
    private String renderTemplate(String templateName, Map<String, Object> data) {
        try {
            Template template = freemarkerConfig.getTemplate(templateName);
            StringWriter writer = new StringWriter();
            template.process(data, writer);
            return writer.toString();
        } catch (Exception e) {
            throw new CodeGenException("模板渲染失败: " + templateName, e);
        }
    }
    
    /**
     * AI生成复杂业务逻辑
     */
    private String generateBusinessLogic(EntitySpec entity, TechnicalSpec spec) {
        
        String prompt = """
            根据以下业务需求，生成Service层中标准CRUD之外的业务逻辑代码。
            
            模块：%s
            实体：%s (%s)
            字段：%s
            
            业务规则：
            %s
            
            已有功能：
            %s
            
            要求：
            1. 实现所有列出的业务规则
            2. 使用Spring事务管理@Transactional
            3. 异常处理使用BusinessException
            4. 使用LambdaQueryWrapper构建复杂查询
            5. 返回类型统一使用Result<T>
            6. 只输出业务方法的实现代码（不要生成整个类）
            """.formatted(
                spec.getModuleNameCN(),
                entity.getName(),
                entity.getComment(),
                formatFields(entity.getFields()),
                formatBusinessRules(entity.getBusinessRules()),
                formatFeatures(spec.getFeatures())
            );
        
        return llm.generate(prompt);
    }
}

### 3.4 增量代码合并——不覆盖已有修改

```java
/**
 * 增量代码合并引擎
 * 用户手动修改的代码不被AI生成覆盖
 */
@Service
public class IncrementalMergeEngine {
    
    /**
     * 智能合并生成的代码和现有代码
     * 使用Git diff + 三路合并策略
     */
    public MergeResult merge(GeneratedModule generated, String projectPath, 
                              boolean overwriteExisting) {
        
        MergeResult result = new MergeResult();
        
        for (Map.Entry<String, String> file : generated.getAllFiles().entrySet()) {
            Path targetPath = Paths.get(projectPath, 
                generated.getModulePath(), file.getKey());
            
            if (Files.exists(targetPath) && !overwriteExisting) {
                // 文件已存在，执行智能合并
                String existingCode = readFile(targetPath);
                String newCode = file.getValue();
                
                // 使用AI判断是否需要合并
                MergeDecision decision = analyzeMergeDecision(
                    targetPath.toString(), existingCode, newCode
                );
                
                switch (decision.getAction()) {
                    case SKIP -> {
                        // 用户修改过，跳过
                        result.addSkipped(targetPath.toString());
                    }
                    case MERGE -> {
                        // AI合并两个版本
                        String merged = aiMerge(existingCode, newCode, 
                            decision.getReason());
                        writeFile(targetPath, merged);
                        result.addMerged(targetPath.toString());
                    }
                    case OVERWRITE -> {
                        // 用户未修改，直接覆盖
                        writeFile(targetPath, newCode);
                        result.addOverwritten(targetPath.toString());
                    }
                }
            } else {
                // 新文件，直接创建
                Files.createDirectories(targetPath.getParent());
                writeFile(targetPath, newCode);
                result.addCreated(targetPath.toString());
            }
        }
        
        return result;
    }
    
    /**
     * AI分析是否需要合并
     */
    private MergeDecision analyzeMergeDecision(String filePath, 
                                                String existing, String newCode) {
        
        String prompt = """
            分析以下代码文件，判断AI生成的代码应该如何处理。
            
            文件路径：%s
            
            现有代码（用户可能已修改）：
            ```java
            %s
            ```
            
            AI生成的代码：
            ```java
            %s
            ```
            
            请判断：
            1. 用户是否对现有代码做了"有意义的修改"（不是AI生成的注释或空行）
            2. 如果有修改，是否可以将AI生成的新功能合并进去而不丢失用户修改
            
            回复格式：
            {
              "action": "SKIP|MERGE|OVERWRITE",
              "reason": "判断理由"
            }
            """.formatted(filePath, 
                truncateLines(existing, 100),
                truncateLines(newCode, 100));
        
        String response = llm.generate(prompt);
        return parseJSON(response, MergeDecision.class);
    }
}
