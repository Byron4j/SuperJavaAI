# 零基础搭建企业知识库：上传文档 → 自动问答（30 分钟全程录屏式教程）

## 开篇：每个老板都有一个梦想

"把我们公司所有的文档扔进去，然后问什么答什么。"

这大概是 2024 年企业 AI 应用最频繁的需求。客服回复标准、产品说明书、公司规章制度、技术文档...如果能用 AI 自动回答，能省多少人力？

好消息是，**实现这个需求只需要 30 分钟**。不需要训练模型，不需要海量数据，不需要 GPU。

坏消息是...没有坏消息。真的就这么简单。

今天我用 Java + Spring Boot + OpenAI API + 本地向量数据库，从零搭建一个企业知识库系统。每一步都截图式的详细说明，跟着做肯定能跑通。

## 一、先理解原理：RAG 到底怎么做？

RAG（Retrieval-Augmented Generation，检索增强生成）听起来高大上，其实就是四步：

```
Step 1: 把文档切成小块（Chunking）
"公司成立于2020年，总部位于北京。员工人数500人。主营业务..."
    ↓ 切分
["公司成立于2020年，总部位于北京。",
 "员工人数500人。",
 "主营业务是软件开发和技术咨询。"]

Step 2: 把每个小块转成向量（Embedding）
"公司成立于2020年，总部位于北京。" 
    ↓ Embedding API
[0.023, 0.845, -0.412, 0.667, ...]  (1536维向量)

Step 3: 用户提问时，找出最相关的文档块
用户问："公司有多少员工？"
    ↓ 把问题也转成向量
    ↓ 和所有文档块向量算相似度
    ↓ 找到最相似的那块
"员工人数500人。" ← 相似度 0.92

Step 4: 把相关文档块+用户问题一起发给 AI
Prompt = "根据以下参考资料回答问题：
参考资料：员工人数500人。
问题：公司有多少员工？"
    ↓ AI 回复
"公司目前有500名员工。"
```

用 Java 类比：

```java
// RAG 就是给大模型加了个"Ctrl+F 搜索"的能力
// 只不过不是关键词匹配，而是语义相似度匹配

// 传统方式：关键词搜索
String result = document.contains("员工") ? document : null;

// RAG方式：语义搜索
float similarity = cosineSimilarity(
    embedding("公司有多少员工？"),
    embedding("员工人数500人。")
);
if (similarity > 0.8) {
    result = "员工人数500人。";
}
```

```java
// service/DocumentParser.java
package com.example.knowledgebase.service;

import org.apache.tika.Tika;
import org.apache.tika.metadata.Metadata;
import org.apache.tika.parser.AutoDetectParser;
import org.apache.tika.parser.ParseContext;
import org.apache.tika.sax.BodyContentHandler;
import org.springframework.stereotype.Service;
import org.springframework.web.multipart.MultipartFile;

import java.io.InputStream;

@Service
public class DocumentParser {
    
    private final Tika tika = new Tika();
    
    /**
     * 解析各种格式的文档，提取纯文本
     * 支持：PDF, Word, Excel, PPT, HTML, Markdown, TXT 等
     */
    public String parse(MultipartFile file) throws Exception {
        // 方式1：简单方式（适合大多数场景）
        return tika.parseToString(file.getInputStream());
    }
    
    /**
     * 解析并保留元数据（文件作者、创建日期等）
     */
    public ParsedDocument parseWithMetadata(MultipartFile file) throws Exception {
        BodyContentHandler handler = new BodyContentHandler(-1); // 无字符限制
        Metadata metadata = new Metadata();
        AutoDetectParser parser = new AutoDetectParser();
        ParseContext context = new ParseContext();
        
        try (InputStream is = file.getInputStream()) {
            parser.parse(is, handler, metadata, context);
        }
        
        return new ParsedDocument(
            handler.toString(),
            metadata.get("dc:title"),
            metadata.get("dc:creator"),
            metadata.get("Content-Type")
        );
    }
    
    @Data
    @AllArgsConstructor
    public static class ParsedDocument {
        private String content;
        private String title;
        private String author;
        private String contentType;
    }
}
```

## 四、核心组件二：文本分块器

```java
// service/TextChunker.java
package com.example.knowledgebase.service;

import org.springframework.stereotype.Service;
import java.util.*;

/**
 * 把长文本切成小块
 * 
 * 为什么需要分块？
 * 1. 向量模型有输入长度限制（通常 8192 Token）
 * 2. 小块检索更精准（不会把无关内容也塞给 AI）
 * 3. 小块便宜（Token 数量少）
 */
@Service
public class TextChunker {
    
    // 每个块约 500 字（根据 Embedding 模型的 Token 限制调整）
    private static final int CHUNK_SIZE = 500;
    
    // 重叠 50 字（避免切到一句话中间导致语义断裂）
    private static final int OVERLAP = 50;
    
    /**
     * 按句子边界智能切分
     */
    public List<Chunk> chunkBySentence(String text, String documentId) {
        // 1. 按句子分割
        List<String> sentences = splitSentences(text);
        
        // 2. 合并句子成块（保证语义完整性）
        List<Chunk> chunks = new ArrayList<>();
        StringBuilder currentChunk = new StringBuilder();
        int chunkIndex = 0;
        
        for (String sentence : sentences) {
            // 如果加上这个句子会超长，就把当前块保存
            if (currentChunk.length() + sentence.length() > CHUNK_SIZE 
                && currentChunk.length() > 0) {
                
                chunks.add(new Chunk(
                    documentId + "_chunk_" + chunkIndex,
                    documentId,
                    currentChunk.toString().trim(),
                    chunkIndex
                ));
                
                // 保留最后 OVERLAP 字符作为重叠
                String overlap = getLastChars(currentChunk.toString(), OVERLAP);
                currentChunk = new StringBuilder(overlap);
                chunkIndex++;
            }
            
            currentChunk.append(sentence);
        }
        
        // 保存最后一块
        if (currentChunk.length() > 0) {
            chunks.add(new Chunk(
                documentId + "_chunk_" + chunkIndex,
                documentId,
                currentChunk.toString().trim(),
                chunkIndex
            ));
        }
        
        return chunks;
    }
    
    /**
     * 按固定大小分块（简单粗暴，适合英文等按词分界的语言）
     */
    public List<Chunk> chunkBySize(String text, String documentId) {
        List<Chunk> chunks = new ArrayList<>();
        int start = 0;
        int index = 0;
        
        while (start < text.length()) {
            int end = Math.min(start + CHUNK_SIZE, text.length());
            
            // 尽量在句号、换行处断开
            if (end < text.length()) {
                int breakPoint = findBreakPoint(text, end);
                if (breakPoint > start) {
                    end = breakPoint;
                }
            }
            
            String chunkText = text.substring(start, end).trim();
            chunks.add(new Chunk(
                documentId + "_chunk_" + index,
                documentId,
                chunkText,
                index
            ));
            
            start = end - OVERLAP;  // 重叠
            index++;
        }
        
        return chunks;
    }
    
    /**
     * 智能切分句子（支持中英文）
     */
    private List<String> splitSentences(String text) {
        List<String> sentences = new ArrayList<>();
        StringBuilder current = new StringBuilder();
        
        for (char c : text.toCharArray()) {
            current.append(c);
            // 中文句号、问号等；英文句号后跟空格
            if (c == '。' || c == '！' || c == '？' || c == '\n'
                || (c == '.' && (current.length() >= 2))) {
                if (current.length() > 1) {
                    sentences.add(current.toString());
                    current = new StringBuilder();
                }
            }
        }
        if (current.length() > 0) {
            sentences.add(current.toString());
        }
        
        return sentences;
    }
    
    private String getLastChars(String text, int count) {
        if (text.length() <= count) return text;
        return text.substring(text.length() - count);
    }
    
    private int findBreakPoint(String text, int end) {
        // 在 end 前后寻找最合适的断点
        int searchStart = Math.max(end - 50, 0);
        for (int i = end; i > searchStart; i--) {
            char c = text.charAt(i);
            if (c == '。' || c == '！' || c == '？' || c == '\n' || c == '.') {
                return i + 1;
            }
        }
        return end;
    }
    
    @Data
    @AllArgsConstructor
    public static class Chunk {
        private String id;          // 唯一标识
        private String documentId;   // 所属文档
        private String content;      // 文本内容
        private int index;           // 在文档中的序号
    }
}
```

## 五、核心组件三：Embedding 服务

```java
// service/EmbeddingService.java
package com.example.knowledgebase.service;

import com.openai.client.OpenAIClient;
import com.openai.models.*;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.stereotype.Service;

import java.util.*;

@Service
public class EmbeddingService {
    
    @Autowired
    private OpenAIClient openAIClient;
    
    // 使用 OpenAI 的 text-embedding-3-small 模型
    // 1536维向量，$0.02/百万token，非常便宜
    private static final String EMBEDDING_MODEL = "text-embedding-3-small";
    
    /**
     * 将文本转为向量
     */
    public float[] embed(String text) {
        EmbeddingCreateParams params = EmbeddingCreateParams.builder()
            .model(EMBEDDING_MODEL)
            .input(text)
            .build();
        
        Embedding embedding = openAIClient.embeddings().create(params);
        
        // 将 Double 转成 float（节省内存）
        List<Double> embeddingList = embedding.data().get(0).embedding();
        float[] vector = new float[embeddingList.size()];
        for (int i = 0; i < vector.length; i++) {
            vector[i] = embeddingList.get(i).floatValue();
        }
        return vector;
    }
    
    /**
     * 批量转为向量（一次 API 调用生成多个 Embedding，省钱）
     */
    public List<float[]> embedBatch(List<String> texts) {
        EmbeddingCreateParams params = EmbeddingCreateParams.builder()
            .model(EMBEDDING_MODEL)
            .inputOfStrings(texts)
            .build();
        
        Embedding embedding = openAIClient.embeddings().create(params);
        
        List<float[]> vectors = new ArrayList<>();
        for (EmbeddingData data : embedding.data()) {
            List<Double> embeddingList = data.embedding();
            float[] vector = new float[embeddingList.size()];
            for (int i = 0; i < vector.length; i++) {
                vector[i] = embeddingList.get(i).floatValue();
            }
            vectors.add(vector);
        }
        return vectors;
    }
    
    /**
     * 计算两个向量的余弦相似度
     */
    public static float cosineSimilarity(float[] a, float[] b) {
        float dotProduct = 0;
        float normA = 0;
        float normB = 0;
        
        for (int i = 0; i < a.length; i++) {
            dotProduct += a[i] * b[i];
            normA += a[i] * a[i];
            normB += b[i] * b[i];
        }
        
        if (normA == 0 || normB == 0) return 0;
        return dotProduct / (float) (Math.sqrt(normA) * Math.sqrt(normB));
    }
}
```

## 六、核心组件四：向量存储与检索

```java
// service/VectorStore.java
package com.example.knowledgebase.service;

import org.springframework.stereotype.Component;
import java.util.*;
import java.util.concurrent.ConcurrentHashMap;

/**
 * 简单的内存向量存储
 * 生产环境建议使用 Milvus / Pinecone / Weaviate / Qdrant
 */
@Component
public class VectorStore {
    
    // 向量存储：chunkId → 向量
    private final Map<String, float[]> vectors = new ConcurrentHashMap<>();
    
    // 文本存储：chunkId → Chunk
    private final Map<String, TextChunker.Chunk> chunks = new ConcurrentHashMap<>();
    
    /**
     * 存储文档块及其向量
     */
    public void store(TextChunker.Chunk chunk, float[] vector) {
        vectors.put(chunk.getId(), vector);
        chunks.put(chunk.getId(), chunk);
    }
    
    /**
     * 批量存储
     */
    public void storeBatch(List<TextChunker.Chunk> chunksList, List<float[]> vectorsList) {
        for (int i = 0; i < chunksList.size(); i++) {
            store(chunksList.get(i), vectorsList.get(i));
        }
    }
    
    /**
     * 搜索最相似的 K 个文档块
     */
    public List<SearchResult> search(float[] queryVector, int topK) {
        // 计算所有文档块的相似度
        PriorityQueue<SearchResult> heap = new PriorityQueue<>(
            Comparator.comparingDouble(SearchResult::getSimilarity)
        );
        
        for (Map.Entry<String, float[]> entry : vectors.entrySet()) {
            String chunkId = entry.getKey();
            float[] chunkVector = entry.getValue();
            
            float similarity = EmbeddingService.cosineSimilarity(queryVector, chunkVector);
            
            TextChunker.Chunk chunk = chunks.get(chunkId);
            if (chunk != null) {
                SearchResult result = new SearchResult(chunk, similarity);
                heap.offer(result);
                
                // 只保留 Top-K
                if (heap.size() > topK) {
                    heap.poll();
                }
            }
        }
        
        // 按相似度高到低排序
        List<SearchResult> results = new ArrayList<>(heap);
        results.sort((a, b) -> Float.compare(b.similarity, a.similarity));
        
        return results;
    }
    
    /**
     * 获取文档块数量
     */
    public int size() {
        return vectors.size();
    }
    
    /**
     * 清空所有数据
     */
    public void clear() {
        vectors.clear();
        chunks.clear();
    }
    
    @Data
    @AllArgsConstructor
    public static class SearchResult {
        private TextChunker.Chunk chunk;
        private float similarity;
    }
}
```

## 七、核心组件五：RAG 问答服务

```java
// service/RAGService.java
package com.example.knowledgebase.service;

import com.openai.client.OpenAIClient;
import com.openai.models.*;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.stereotype.Service;
import org.springframework.web.multipart.MultipartFile;

import java.util.*;
import java.util.stream.Collectors;

@Service
public class RAGService {
    
    @Autowired private DocumentParser documentParser;
    @Autowired private TextChunker textChunker;
    @Autowired private EmbeddingService embeddingService;
    @Autowired private VectorStore vectorStore;
    @Autowired private OpenAIClient openAIClient;
    
    @Value("${openai.model}") private String model;
    
    /**
     * 上传文档并建立索引
     */
    public Map<String, Object> uploadDocument(MultipartFile file) {
        try {
            // 1. 解析文档
            String content = documentParser.parse(file);
            String documentId = UUID.randomUUID().toString().substring(0, 8);
            
            // 2. 文本分块
            List<TextChunker.Chunk> chunks = textChunker.chunkBySentence(content, documentId);
            
            // 3. 提取所有块的文本
            List<String> chunkTexts = chunks.stream()
                .map(TextChunker.Chunk::getContent)
                .collect(Collectors.toList());
            
            // 4. 批量生成 Embedding
            List<float[]> vectors = embeddingService.embedBatch(chunkTexts);
            
            // 5. 存入向量数据库
            vectorStore.storeBatch(chunks, vectors);
            
            return Map.of(
                "success", true,
                "documentId", documentId,
                "fileName", file.getOriginalFilename(),
                "chunks", chunks.size(),
                "totalChars", content.length()
            );
            
        } catch (Exception e) {
            return Map.of(
                "success", false,
                "error", e.getMessage()
            );
        }
    }
    
    /**
     * RAG 问答
     */
    public Map<String, Object> ask(String question) {
        // 1. 把问题转成向量
        float[] questionVector = embeddingService.embed(question);
        
        // 2. 检索最相关的文档块（Top 5）
        List<VectorStore.SearchResult> results = vectorStore.search(questionVector, 5);
        
        // 3. 拼接上下文
        StringBuilder context = new StringBuilder();
        for (int i = 0; i < results.size(); i++) {
            VectorStore.SearchResult result = results.get(i);
            context.append(String.format(
                "[参考资料 %d] (相关度: %.2f)\n%s\n\n",
                i + 1,
                result.getSimilarity(),
                result.getChunk().getContent()
            ));
        }
        
        // 4. 构建 Prompt
        String prompt = promptTemplate
            .replace("{context}", context.toString())
            .replace("{question}", question);
        
        // 5. 调用大模型
        ChatCompletionCreateParams params = ChatCompletionCreateParams.builder()
            .model(ChatModel.of(model))
            .addUserMessage(prompt)
            .maxTokens(500)
            .temperature(0.3)  // 知识库回答用低温度，更准确
            .build();
        
        ChatCompletion completion = openAIClient.chat().completions().create(params);
        String answer = completion.choices().get(0).message().content().orElse("无法生成回答");
        
        // 6. 构建返回结果
        List<Map<String, Object>> sources = results.stream()
            .map(r -> Map.of(
                "content", (Object) r.getChunk().getContent().substring(0, 
                    Math.min(100, r.getChunk().getContent().length())),
                "similarity", r.getSimilarity(),
                "documentId", r.getChunk().getDocumentId()
            ))
            .collect(Collectors.toList());
        
        return Map.of(
            "question", question,
            "answer", answer,
            "sources", sources
        );
    }
    
    private static final String promptTemplate = """
        你是一个企业知识库助手。请根据以下参考资料回答用户问题。
        
        规则：
        1. 只根据参考资料回答，不要使用你自己的知识
        2. 如果参考资料中没有相关信息，请明确说"根据现有资料无法回答"
        3. 回答时注明引用的资料编号
        4. 回答简洁、准确、专业
        
        参考资料：
        {context}
        
        用户问题：{question}
        
        请回答：
        """;
}
```

## 八、REST API

```java
// controller/KnowledgeBaseController.java
package com.example.knowledgebase.controller;

import com.example.knowledgebase.service.RAGService;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.web.bind.annotation.*;
import org.springframework.web.multipart.MultipartFile;

import java.util.Map;

@RestController
@RequestMapping("/api/knowledge-base")
public class KnowledgeBaseController {
    
    @Autowired
    private RAGService ragService;
    
    /**
     * 上传文档
     */
    @PostMapping("/upload")
    public Map<String, Object> upload(@RequestParam("file") MultipartFile file) {
        return ragService.uploadDocument(file);
    }
    
    /**
     * 问答
     */
    @PostMapping("/ask")
    public Map<String, Object> ask(@RequestBody Map<String, String> request) {
        String question = request.get("question");
        if (question == null || question.trim().isEmpty()) {
            return Map.of("error", "问题不能为空");
        }
        return ragService.ask(question);
    }
    
    /**
     * 知识库状态
     */
    @GetMapping("/status")
    public Map<String, Object> status() {
        return Map.of("totalChunks", 0); // 可以添加更多统计
    }
}
```

## 十一、进阶优化建议

### 1. 混合检索（Hybrid Search）

```java
// 结合关键词检索和向量检索
public List<SearchResult> hybridSearch(String question, int topK) {
    // 向量检索（语义匹配）
    float[] queryVector = embeddingService.embed(question);
    List<SearchResult> semanticResults = vectorStore.search(queryVector, topK * 2);
    
    // 关键词检索（精确匹配）
    List<SearchResult> keywordResults = keywordSearch(question, topK);
    
    // 融合排序（Reciprocal Rank Fusion）
    Map<String, Double> scores = new HashMap<>();
    for (int i = 0; i < semanticResults.size(); i++) {
        scores.merge(semanticResults.get(i).getChunk().getId(), 
            1.0 / (i + 60), Double::sum);
    }
    for (int i = 0; i < keywordResults.size(); i++) {
        scores.merge(keywordResults.get(i).getChunk().getId(), 
            1.0 / (i + 60), Double::sum);
    }
    
    // 按综合分数排序
    return scores.entrySet().stream()
        .sorted(Map.Entry.<String, Double>comparingByValue().reversed())
        .limit(topK)
        .map(e -> findChunkById(e.getKey()))
        .collect(Collectors.toList());
}
```
