## 一、五个部门，五套 AI Key，五本糊涂账

这是我去年加入一家 500 人规模 SaaS 公司时面临的现状：

### 3.2 模型管理服务

```java
@Service
@Slf4j
public class ModelManagementService {

    @Autowired
    private ModelRepository modelRepository;
    
    @Autowired
    private ModelHealthChecker healthChecker;
    
    @Autowired
    private ModelRegistryEventPublisher eventPublisher;

    /**
     * 注册新模型
     */
    @Transactional
    public ModelEntity registerModel(ModelRegistrationRequest request) {
        // 1. 验证唯一性
        if (modelRepository.existsByModelCode(request.getModelCode())) {
            throw new DuplicateModelException("模型已存在: " + request.getModelCode());
        }
        
        // 2. 加密凭据
        ModelEntity entity = new ModelEntity();
        BeanUtils.copyProperties(request, entity);
        entity.setEncryptedCredential(encryptCredential(request.getCredential()));
        entity.setStatus(ModelStatus.MAINTENANCE);  // 注册后先维护态
        entity.setCreatedAt(LocalDateTime.now());
        entity.setCreatedBy(AuditContextHolder.getCurrentUser());
        
        ModelEntity saved = modelRepository.save(entity);
        
        // 3. 健康检查
        boolean healthy = healthChecker.check(saved);
        if (healthy) {
            saved.setStatus(ModelStatus.ACTIVE);
            modelRepository.save(saved);
            log.info("✅ 模型注册成功并激活: {}", saved.getModelCode());
        } else {
            log.warn("⚠️ 模型注册成功但健康检查失败: {}", saved.getModelCode());
        }
        
        // 4. 发布事件
        eventPublisher.publishModelRegistered(saved);
        
        return saved;
    }
    
    /**
     * 获取场景可用模型列表（带排序）
     */
    public List<ModelDTO> getAvailableModels(String scene, String tenantId) {
        return modelRepository.findActiveByScene(scene, tenantId).stream()
            .sorted(Comparator.comparingInt(ModelEntity::getPriority))
            .map(this::toDTO)
            .collect(Collectors.toList());
    }
    
    /**
     * 更新模型状态
     */
    @Transactional
    public void updateStatus(Long id, ModelStatus newStatus) {
        ModelEntity model = modelRepository.findById(id)
            .orElseThrow(() -> new ModelNotFoundException(id));
        
        ModelStatus oldStatus = model.getStatus();
        model.setStatus(newStatus);
        model.setUpdatedAt(LocalDateTime.now());
        modelRepository.save(model);
        
        log.info("模型 {} 状态变更: {} → {}", model.getModelCode(), oldStatus, newStatus);
        eventPublisher.publishModelStatusChanged(model, oldStatus, newStatus);
    }
}
```

### 3.3 定时健康检查

```java
@Component
public class ModelHealthChecker {

    @Autowired
    private ModelRepository modelRepository;
    
    @Autowired
    private AlertService alertService;
    
    /** 连续失败计数 */
    private final Map<String, AtomicInteger> failureCounters = new ConcurrentHashMap<>();

    /**
     * 每 30 秒检查所有活跃模型
     */
    @Scheduled(fixedRate = 30000)
    public void checkAllActiveModels() {
        List<ModelEntity> activeModels = modelRepository.findByStatus(ModelStatus.ACTIVE);
        
        for (ModelEntity model : activeModels) {
            boolean healthy = check(model);
            
            if (!healthy) {
                int failures = failureCounters
                    .computeIfAbsent(model.getModelCode(), k -> new AtomicInteger(0))
                    .incrementAndGet();
                
                if (failures == 3) {
                    // 连续 3 次失败，自动下线
                    model.setStatus(ModelStatus.MAINTENANCE);
                    modelRepository.save(model);
                    
                    alertService.sendAlert(AlertLevel.HIGH,
                        String.format("模型 %s 连续健康检查失败，已自动下线", 
                            model.getModelCode()));
                    
                    failureCounters.remove(model.getModelCode());
                }
            } else {
                failureCounters.remove(model.getModelCode());
            }
        }
    }
    
    private boolean check(ModelEntity model) {
        try {
            String healthUrl = model.getEndpoint() + "/health";
            RestTemplate rt = new RestTemplate();
            ResponseEntity<String> resp = rt.getForEntity(healthUrl, String.class);
            return resp.getStatusCode().is2xxSuccessful();
        } catch (Exception e) {
            log.debug("模型 {} 健康检查失败: {}", model.getModelCode(), e.getMessage());
            return false;
        }
    }
}
```

### 4.2 API Key 认证过滤器

```java
@Component
public class AIApiKeyAuthFilter extends OncePerRequestFilter {

    @Autowired
    private ApiKeyService apiKeyService;
    
    @Autowired
    private QuotaService quotaService;

    private static final Pattern API_KEY_PATTERN = Pattern.compile("^sk-ai[a-z0-9]{32}$");
    private static final List<String> PUBLIC_ENDPOINTS = List.of(
        "/api/v1/health", "/api/v1/models/public");

    @Override
    protected void doFilterInternal(HttpServletRequest request, 
            HttpServletResponse response, FilterChain chain) {
        
        String path = request.getRequestURI();
        
        // 公开接口跳过
        if (PUBLIC_ENDPOINTS.stream().anyMatch(path::startsWith)) {
            chain.doFilter(request, response);
            return;
        }
        
        // 提取 API Key
        String apiKey = request.getHeader("X-API-Key");
        if (apiKey == null) {
            apiKey = request.getHeader("Authorization");
            if (apiKey != null && apiKey.startsWith("Bearer ")) {
                apiKey = apiKey.substring(7);
            }
        }
        
        if (apiKey == null || !API_KEY_PATTERN.matcher(apiKey).matches()) {
            response.setStatus(401);
            response.getWriter().write("{\"error\":\"Invalid or missing API Key\"}");
            return;
        }
        
        // 验证 API Key
        ApiKeyInfo keyInfo = apiKeyService.validate(apiKey);
        if (keyInfo == null) {
            response.setStatus(401);
            response.getWriter().write("{\"error\":\"API Key invalid or expired\"}");
            return;
        }
        
        // 检查配额
        if (!quotaService.checkAndDeduct(keyInfo.getUserId(), request)) {
            response.setStatus(429);
            response.getWriter().write(
                "{\"error\":\"Quota exceeded. Daily limit reached.\"}");
            return;
        }
        
        // 设置上下文
        AuditContextHolder.setCurrentUser(keyInfo);
        
        chain.doFilter(request, response);
    }
}
```

### 4.3 配额服务

```java
@Service
public class QuotaService {

    @Autowired
    private StringRedisTemplate redisTemplate;
    
    @Autowired
    private ModelManagementService modelService;
    
    @Autowired
    private AlertService alertService;

    /**
     * 检查并扣减配额
     * 
     * 使用 Redis Lua 脚本保证原子性
     */
    public boolean checkAndDeduct(Long userId, HttpServletRequest request) {
        String dailyTokenKey = "quota:daily_tokens:" + userId + ":" + LocalDate.now();
        String dailyCostKey = "quota:daily_cost:" + userId + ":" + LocalDate.now();
        
        // 预估本次调用 Token 消耗（简单场景用请求体大小估算）
        long estimatedTokens = estimateTokens(request);
        
        Long dailyLimit = getUserTokenLimit(userId);
        Long costLimit = getUserCostLimit(userId);
        
        // Lua 脚本原子操作
        String luaScript = """
            local tokenKey = KEYS[1]
            local costKey = KEYS[2]
            local tokens = tonumber(ARGV[1])
            local tokenLimit = tonumber(ARGV[2])
            local costLimit = tonumber(ARGV[3])
            
            local currentTokens = tonumber(redis.call('GET', tokenKey) or '0')
            local currentCost = tonumber(redis.call('GET', costKey) or '0')
            
            if currentTokens + tokens > tokenLimit then
                return 0  -- Token 超配额
            end
            
            redis.call('INCRBY', tokenKey, tokens)
            redis.call('EXPIRE', tokenKey, 86400)
            return 1
        """;
        
        Long result = redisTemplate.execute(
            new DefaultRedisScript<>(luaScript, Long.class),
            List.of(dailyTokenKey, dailyCostKey),
            String.valueOf(estimatedTokens),
            String.valueOf(dailyLimit),
            String.valueOf(costLimit)
        );
        
        if (result != null && result == 0) {
            // 超配额告警
            alertService.sendQuotaAlert(userId, dailyTokenKey);
            return false;
        }
        
        return true;
    }
    
    /**
     * 事后精确扣减（拿到实际 Token 数后）
     */
    public void deductActualTokens(Long userId, int actualTokens, double actualCost) {
        String dailyTokenKey = "quota:daily_tokens:" + userId + ":" + LocalDate.now();
        String dailyCostKey = "quota:daily_cost:" + userId + ":" + LocalDate.now();
        
        // 实际用量可能跟预估值不同，这里直接 SET 实际值
        redisTemplate.opsForValue().increment(dailyTokenKey, actualTokens);
        redisTemplate.opsForValue().increment(dailyCostKey, (long)(actualCost * 100));
    }
    
    /**
     * 查询用户今日用量
     */
    public QuotaUsageDTO getTodayUsage(Long userId) {
        String tokenKey = "quota:daily_tokens:" + userId + ":" + LocalDate.now();
        String costKey = "quota:daily_cost:" + userId + ":" + LocalDate.now();
        
        long tokens = Long.parseLong(
            redisTemplate.opsForValue().get(tokenKey) != null 
                ? redisTemplate.opsForValue().get(tokenKey) : "0");
        long costCents = Long.parseLong(
            redisTemplate.opsForValue().get(costKey) != null 
                ? redisTemplate.opsForValue().get(costKey) : "0");
        
        return QuotaUsageDTO.builder()
            .userId(userId)
            .todayTokens(tokens)
            .todayCost(costCents / 100.0)
            .tokenLimit(getUserTokenLimit(userId))
            .costLimit(getUserCostLimit(userId))
            .tokenUsagePercent((double) tokens / getUserTokenLimit(userId) * 100)
            .build();
    }
    
    private long estimateTokens(HttpServletRequest request) {
        int contentLength = request.getContentLength();
        return contentLength > 0 ? contentLength / 4 : 500;
    }
}
```

### 5.2 限流降级实现

```java
@Component
public class AiGatewayFilter implements GlobalFilter, Ordered {

    @Autowired
    private ModelManagementService modelService;
    
    @Autowired
    private ModelRouter modelRouter;
    
    @Autowired
    private AuditLogger auditLogger;

    @Override
    public Mono<Void> filter(ServerWebExchange exchange, GatewayFilterChain chain) {
        ServerHttpRequest request = exchange.getRequest();
        String path = request.getPath().value();
        
        // 非 AI 调用请求直接放过
        if (!path.startsWith("/api/v1/chat") && !path.startsWith("/api/v1/embeddings")) {
            return chain.filter(exchange);
        }
        
        // 解析请求体和场景
        return request.getBody().collectList().flatMap(dataBuffers -> {
            String body = dataBuffers.stream()
                .map(b -> {
                    byte[] bytes = new byte[b.readableByteCount()];
                    b.read(bytes);
                    return new String(bytes);
                })
                .collect(Collectors.joining());
            
            ChatRequest chatRequest = JSON.parseObject(body, ChatRequest.class);
            
            // 场景感知路由
            String scene = chatRequest.getScene() != null 
                ? chatRequest.getScene() : "general_chat";
            
            // 获取场景可用模型
            List<ModelDTO> availableModels = modelService.getAvailableModels(
                scene, AuditContextHolder.getTenantId());
            
            if (availableModels.isEmpty()) {
                return errorResponse(exchange, 503, "场景 " + scene + " 无可用模型");
            }
            
            // 传递模型信息到下游
            exchange.getAttributes().put("available_models", availableModels);
            exchange.getAttributes().put("scene", scene);
            
            return chain.filter(exchange);
        });
    }
    
    @Override
    public int getOrder() {
        return -100;  // 高优先级
    }
}
```

```java
@RestController
@RequestMapping("/api/admin/batch")
@PreAuthorize("hasRole('ADMIN')")
public class BatchOperationController {

    @Autowired
    private ModelManagementService modelService;

    /**
     * 批量启用/禁用用户 AI 权限
     */
    @PostMapping("/users/toggle-ai-access")
    public BatchResult toggleUserAIAccess(@RequestBody List<Long> userIds, 
            @RequestParam boolean enable) {
        int success = 0, failed = 0;
        for (Long id : userIds) {
            try {
                modelService.setUserAIAccess(id, enable);
                success++;
            } catch (Exception e) {
                failed++;
            }
        }
        return new BatchResult(success, failed);
    }
    
    /**
     * 批量调整配额
     */
    @PostMapping("/users/adjust-quota")
    public BatchResult adjustQuota(@RequestBody QuotaAdjustmentRequest request) {
        return quotaService.batchAdjust(request.getUserIds(), 
            request.getTokenLimit(), request.getCostLimit());
    }
}
```

```java
@Component
public class AIMetricsCollector {

    private final MeterRegistry registry;
    
    // 模型调用计数器（按模型、场景分组）
    private final Map<String, Counter> modelCallCounters = new ConcurrentHashMap<>();
    
    // 模型延迟直方图
    private final Map<String, Timer> modelLatencyTimers = new ConcurrentHashMap<>();
    
    // Token 消耗计量器
    private final Counter totalTokensCounter;
    private final Counter totalCostCounter;
    
    public AIMetricsCollector(MeterRegistry registry) {
        this.registry = registry;
        this.totalTokensCounter = Counter.builder("ai.tokens.total")
            .description("Total tokens consumed")
            .register(registry);
        this.totalCostCounter = Counter.builder("ai.cost.total")
            .description("Total cost in RMB cents")
            .baseUnit("cents")
            .register(registry);
    }

    public void recordModelCall(String modelCode, String scene, 
            int tokens, double cost, long latencyMs) {
        // 按模型+场景计数
        String key = modelCode + ":" + scene;
        modelCallCounters.computeIfAbsent(key, k ->
            Counter.builder("ai.model.calls")
                .tag("model", modelCode)
                .tag("scene", scene)
                .register(registry)
        ).increment();
        
        // 延迟记录
        modelLatencyTimers.computeIfAbsent(key, k ->
            Timer.builder("ai.model.latency")
                .tag("model", modelCode)
                .tag("scene", scene)
                .register(registry)
        ).record(latencyMs, TimeUnit.MILLISECONDS);
        
        // 全局计数
        totalTokensCounter.increment(tokens);
        totalCostCounter.increment((long)(cost * 100));
    }
}
```
