## 引言：模型选型的本质与方法论

模型选型不是"选最好的"，而是**在约束条件下选最合适的**。与硬件选型类似，AI模型选型需要在能力、成本、延迟、隐私、生态等多个维度间取得平衡。

```
模型选型的本质：

错误认知：
"选能力最强的模型（GPT-5.5/Claude 4）就对了"

正确认知：
在以下约束下，选择性价比最高的模型：
┌─────────────────────────────────────────┐
│ 约束条件：                                │
│ - 预算：每月$50 vs $500 vs $5000         │
│ - 延迟：<100ms vs <2s vs <10s              │
│ - 隐私：公开数据 vs 商业机密 vs 国家机密    │
│ - 语言：英文 vs 中文 vs 多语言              │
│ - 团队：1人 vs 10人 vs 100人               │
│ - 场景：算法 vs 业务 vs 架构 vs 运维        │
└─────────────────────────────────────────┘

核心洞察：
没有"最好的模型"，只有"最适合的模型"。
```

**核心认知**：模型选型是一种**多目标优化问题**，需要在能力、成本、速度、隐私等多个维度间找到帕累托最优解。

### 2. 成本效益分析模型

```python
class CostBenefitAnalyzer:
    """成本效益分析器"""
    
    def __init__(self):
        self.pricing = {
            'gpt-5.5': {'input': 0.03, 'output': 0.06},  # $/1K tokens
            'claude-4': {'input': 0.03, 'output': 0.15},
            'deepseek-v4': {'input': 0.004, 'output': 0.016},
            'qwen3-coder': {'input': 0.006, 'output': 0.02},
            'kimi-k2.6': {'input': 0.015, 'output': 0.06}
        }
    
    def calculate_cost(self, model_name, input_tokens, output_tokens, 
                       requests_per_day=1000):
        """计算月度成本"""
        price = self.pricing.get(model_name, {'input': 0.01, 'output': 0.03})
        
        daily_cost = (
            input_tokens * price['input'] / 1000 +
            output_tokens * price['output'] / 1000
        ) * requests_per_day
        
        monthly_cost = daily_cost * 30
        
        return {
            'daily': daily_cost,
            'monthly': monthly_cost,
            'yearly': monthly_cost * 12
        }
    
    def calculate_roi(self, model_name, capability_score, cost, 
                      developer_hourly_rate=50):
        """计算ROI"""
        # 假设：模型辅助提升30%效率
        efficiency_gain = 0.3
        hours_saved_per_day = 8 * efficiency_gain  # 假设每天工作8小时
        value_per_day = hours_saved_per_day * developer_hourly_rate
        
        roi = (value_per_day * 30 - cost['monthly']) / cost['monthly']
        
        return {
            'roi': roi,
            'value_per_month': value_per_day * 30,
            'cost_per_month': cost['monthly'],
            'net_value': value_per_day * 30 - cost['monthly']
        }
    
    def recommend(self, requirements, budget):
        """推荐最优模型"""
        candidates = []
        
        for model_name in self.pricing.keys():
            # 获取能力评分
            capability = get_capability_score(model_name, requirements['scenario'])
            
            # 计算成本
            cost = self.calculate_cost(
                model_name,
                requirements['avg_input_tokens'],
                requirements['avg_output_tokens'],
                requirements['requests_per_day']
            )
            
            # 计算ROI
            roi = self.calculate_roi(model_name, capability, cost)
            
            candidates.append({
                'model': model_name,
                'capability': capability,
                'cost': cost,
                'roi': roi['roi'],
                'within_budget': cost['monthly'] <= budget
            })
        
        # 筛选预算内的模型
        affordable = [c for c in candidates if c['within_budget']]
        
        if not affordable:
            # 如果都超预算，选ROI最高的
            return max(candidates, key=lambda x: x['roi'])
        
        # 在预算内选ROI最高的
        return max(affordable, key=lambda x: x['roi'])
```

### 3. 延迟与质量权衡模型

```python
class LatencyQualityTradeoff:
    """延迟与质量权衡模型"""
    
    def __init__(self):
        self.latency_profiles = {
            'gpt-5.5': {'ttft': 2.0, 'tps': 50},  # TTFT: Time To First Token
            'claude-4': {'ttft': 1.5, 'tps': 45},
            'deepseek-v4': {'ttft': 1.0, 'tps': 40},
            'qwen3-coder': {'ttft': 0.8, 'tps': 60},
            'kimi-k2.6': {'ttft': 1.2, 'tps': 45}
        }
    
    def calculate_total_latency(self, model_name, input_tokens, output_tokens):
        """计算总延迟"""
        profile = self.latency_profiles[model_name]
        
        # 总延迟 = 首token延迟 + 生成延迟
        ttft = profile['ttft']
        generation_time = output_tokens / profile['tps']
        
        return ttft + generation_time
    
    def find_optimal(self, requirements):
        """找到最优权衡点"""
        candidates = []
        
        for model_name, profile in self.latency_profiles.items():
            latency = self.calculate_total_latency(
                model_name,
                requirements['input_tokens'],
                requirements['output_tokens']
            )
            
            quality = get_quality_score(model_name, requirements['scenario'])
            
            # 帕累托前沿判断
            dominated = False
            for other in candidates:
                if (other['latency'] <= latency and other['quality'] >= quality):
                    dominated = True
                    break
            
            if not dominated:
                # 移除被当前模型支配的候选
                candidates = [
                    c for c in candidates 
                    if not (c['latency'] >= latency and c['quality'] <= quality)
                ]
                candidates.append({
                    'model': model_name,
                    'latency': latency,
                    'quality': quality
                })
        
        return candidates
```

class DynamicModelRouter:
    """动态模型路由器"""
    
    def __init__(self):
        self.models = {
            'deepseek-v4': {'cost_per_1k': 0.004, 'capability': 9.14},
            'gpt-5.5': {'cost_per_1k': 0.045, 'capability': 9.14},
            'claude-4': {'cost_per_1k': 0.09, 'capability': 9.14},
            'codegeex': {'cost_per_1k': 0, 'capability': 8.14}
        }
        
        self.scenario_thresholds = {
            'algorithm': 0.9,      # 算法需要高能力
            'business': 0.85,      # 业务开发中等
            'architecture': 0.9,   # 架构设计高能力
            'ops': 0.8,            # 运维脚本较低
            'review': 0.85,        # 代码审查中等
            'learning': 0.8        # 学习新技术较低
        }
    
    def route(self, scenario, prompt, budget_constraint=None):
        """动态路由到最合适的模型"""
        
        # 1. 确定场景要求
        required_capability = self.scenario_thresholds.get(scenario, 0.85)
        
        # 2. 筛选满足能力要求的模型
        candidates = [
            (name, info) for name, info in self.models.items()
            if info['capability'] >= required_capability
        ]
        
        if not candidates:
            # 如果没有满足要求的，选能力最强的
            candidates = [max(self.models.items(), key=lambda x: x[1]['capability'])]
        
        # 3. 在预算约束下选最便宜的
        if budget_constraint:
            affordable = [
                (name, info) for name, info in candidates
                if info['cost_per_1k'] <= budget_constraint
            ]
            if affordable:
                candidates = affordable
        
        # 4. 选择成本最低的
        selected = min(candidates, key=lambda x: x[1]['cost_per_1k'])
        
        return selected[0]
    
    def optimize_batch(self, requests):
        """批量优化（合并相似请求）"""
        # 按场景分组
        grouped = defaultdict(list)
        for req in requests:
            grouped[req['scenario']].append(req)
        
        # 为每组选择模型
        results = []
        for scenario, reqs in grouped.items():
            model = self.route(scenario, None)
            results.append({
                'model': model,
                'requests': reqs,
                'estimated_cost': len(reqs) * self.models[model]['cost_per_1k']
            })
        
        return results

# 使用示例

        # 评分标准
        scores = {}
        for model_name, result in results.items():
            score = 0
            
            # 长度适中
            if 100 < len(result) < 1000:
                score += 1
            
            # 包含代码块
            if '```' in result:
                score += 1
            
            # 无错误信息
            if 'error' not in result.lower():
                score += 1
            
            # 模型权重（根据场景）
            weights = {
                'algorithm': {'deepseek': 1.2, 'claude': 1.1, 'gpt': 1.0},
                'business': {'deepseek': 1.2, 'gpt': 1.1, 'claude': 1.0}
            }
            
            weight = weights.get(scenario, {}).get(model_name, 1.0)
            scores[model_name] = score * weight
        
        # 选择得分最高的
        best_model = max(scores, key=scores.get)
        return results[best_model]
    
    def cascade_generate(self, prompt, scenario, budget):
        """级联生成（从便宜到贵）"""

```python
class ModelSelectionABTest:
    def __init__(self):
        self.experiments = {}
    
    def create_experiment(self, name, model_a, model_b, 
                          evaluation_criteria):
        """创建A/B测试实验"""
        self.experiments[name] = {
            'model_a': model_a,
            'model_b': model_b,
            'criteria': evaluation_criteria,
            'results': {'A': [], 'B': []}
        }
    
    def run_test(self, experiment_name, tasks, sample_size=100):
        """运行A/B测试"""
        exp = self.experiments[experiment_name]
        
        for i in range(sample_size):
            task = random.choice(tasks)
            
            # 随机分配
            variant = 'A' if random.random() < 0.5 else 'B'
            model = exp[f'model_{variant.lower()}']
            
            # 生成结果
            start_time = time.time()
            result = model.generate(task['prompt'])
            latency = time.time() - start_time
            
            # 评估结果
            scores = self.evaluate_result(result, task, exp['criteria'])
            scores['latency'] = latency
            
            exp['results'][variant].append(scores)
        
        return self.analyze_results(experiment_name)
    
    def evaluate_result(self, result, task, criteria):
        """评估生成结果"""
        scores = {}
        
        for criterion, weight in criteria.items():
            if criterion == 'correctness':
                scores[criterion] = self.check_correctness(result, task['expected'])
            elif criterion == 'completeness':
                scores[criterion] = self.check_completeness(result, task['requirements'])
            elif criterion == 'style':
                scores[criterion] = self.check_style(result, task['language'])
            elif criterion == 'safety':
                scores[criterion] = self.check_safety(result)
        
        return scores
    
    def analyze_results(self, experiment_name):
        """分析A/B测试结果"""
        exp = self.experiments[experiment_name]
        results_a = exp['results']['A']
        results_b = exp['results']['B']
        
        analysis = {}
        
        for criterion in exp['criteria'].keys():
            scores_a = [r[criterion] for r in results_a]
            scores_b = [r[criterion] for r in results_b]
            
            # t检验
            from scipy import stats
            t_stat, p_value = stats.ttest_ind(scores_a, scores_b)
            
            analysis[criterion] = {
                'A_mean': np.mean(scores_a),
                'B_mean': np.mean(scores_b),
                'A_std': np.std(scores_a),
                'B_std': np.std(scores_b),
                't_statistic': t_stat,
                'p_value': p_value,
                'significant': p_value < 0.05,
                'winner': 'A' if np.mean(scores_a) > np.mean(scores_b) else 'B'
            }
        
        return analysis
```

class IntelligentModelRouter:
    """智能模型路由器"""
    
    def __init__(self):
        self.models = self.load_models()
        self.task_classifier = TaskClassifier()
        self.cost_optimizer = CostOptimizer()
        self.performance_tracker = PerformanceTracker()
    
    def route(self, task):

        # 1. 任务分类
        task_type = self.task_classifier.classify(task)
        complexity = self.task_classifier.assess_complexity(task)
        
        # 2. 约束识别
        constraints = {
            'budget': task.get('budget', 'medium'),
            'latency': task.get('latency_requirement', 'normal'),
            'privacy': task.get('privacy_level', 'low'),
            'language': task.get('language', 'english')
        }
        
        # 3. 候选模型筛选
        candidates = self.filter_candidates(task_type, constraints)
        
        # 4. 成本优化
        optimal = self.cost_optimizer.optimize(candidates, task, constraints)
        
        # 5. 性能预测
        predicted_performance = self.performance_tracker.predict(optimal, task)
        
        # 6. 最终决策
        decision = {
            'model': optimal,
            'confidence': predicted_performance['confidence'],
            'estimated_cost': predicted_performance['cost'],
            'estimated_latency': predicted_performance['latency'],
            'fallback': self.get_fallback(optimal)
        }
        
        return decision
    
    def filter_candidates(self, task_type, constraints):
        """根据约束筛选候选模型"""
        candidates = []
        
        for model_name, model_info in self.models.items():

            if constraints['privacy'] == 'high' and not model_info['local_deployable']:
                continue

            # 能力约束
            if model_info['capabilities'].get(task_type, 0) < 7:
                continue
            
            candidates.append((model_name, model_info))
        
        return candidates
    
    def get_fallback(self, primary_model):
        """获取降级方案"""
        fallbacks = {
            'claude-4': 'deepseek-v4',
            'gpt-5.5': 'deepseek-v4',
            'deepseek-v4': 'qwen3-coder',
            'qwen3-coder': 'codegeex'
        }
        return fallbacks.get(primary_model, 'codegeex')

class TaskClassifier:
    """任务分类器"""
    
    def __init__(self):
        self.classifier = self.load_classifier_model()
    
    def classify(self, task):
        """分类任务类型"""
        # 基于任务描述分类
        features = self.extract_features(task['description'])
        
        task_types = [
            'algorithm',
            'business_development',
            'architecture_design',
            'ops_script',
            'code_review',
            'learning'
        ]
        
        probabilities = self.classifier.predict(features)
        
        return task_types[np.argmax(probabilities)]
    
    def assess_complexity(self, task):
        """评估任务复杂度"""
        factors = {
            'description_length': len(task['description']) / 1000,
            'has_constraints': 1 if 'requirement' in task else 0,
            'multiple_languages': 1 if 'translate' in task['description'] else 0,
            'security_sensitive': 1 if any(kw in task['description'] 
                                           for kw in ['security', 'auth', 'encrypt']) else 0
        }
        
        complexity = sum(factors.values()) / len(factors)
        return min(complexity, 1.0)
    
    def extract_features(self, description):
        """提取任务特征"""
        features = {
            'length': len(description),
            'code_blocks': description.count('```'),
            'technical_terms': self.count_technical_terms(description),
            'language_mentions': self.detect_languages(description)
        }
        return features

class CostOptimizer:
    """成本优化器"""
    
    def optimize(self, candidates, task, constraints):
        """优化选择（成本效益最大化）"""
        
        scores = []
        for model_name, model_info in candidates:
            # 能力分
            capability = model_info['capabilities'].get(
                self.get_task_category(task), 0
            )
            
            # 成本分（越低越好，取反）
            max_cost = max(c[1]['cost_per_1k'] for c in candidates)
            cost_score = 1 - (model_info['cost_per_1k'] / max_cost)
            
            # 速度分
            speed_score = model_info['tokens_per_second'] / 100
            
            # 综合得分
            if constraints['budget'] == 'tight':
                score = capability * 0.3 + cost_score * 0.5 + speed_score * 0.2
            elif constraints['budget'] == 'generous':
                score = capability * 0.6 + cost_score * 0.1 + speed_score * 0.3
            else:
                score = capability * 0.4 + cost_score * 0.3 + speed_score * 0.3
            
            scores.append((model_name, score))
        
        # 选择得分最高的
        return max(scores, key=lambda x: x[1])[0]

class PerformanceTracker:
    """性能追踪器"""
    
    def __init__(self):
        self.history = defaultdict(list)
    
    def predict(self, model_name, task):
        """预测模型在该任务上的性能"""
        
        # 基于历史数据预测
        historical = self.history.get((model_name, task.get('type')), [])
        
        if historical:
            avg_latency = np.mean([h['latency'] for h in historical])
            avg_quality = np.mean([h['quality'] for h in historical])
            confidence = min(len(historical) / 10, 1.0)
        else:
            # 无历史数据，使用模型默认性能
            avg_latency = self.get_default_latency(model_name)
            avg_quality = 0.8
            confidence = 0.3
        
        # 获取成本
        cost = self.estimate_cost(model_name, task)
        
        return {
            'latency': avg_latency,
            'quality': avg_quality,
            'cost': cost,
            'confidence': confidence
        }
    
    def record(self, model_name, task, result):
        """记录实际性能"""
        self.history[(model_name, task.get('type'))].append({
            'latency': result['latency'],
            'quality': result['quality'],
            'timestamp': time.time()
        })
