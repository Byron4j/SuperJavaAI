```java
/**
 * 合规规则引擎
 */
@Component
public class ComplianceEngine {
    
    private final List<ComplianceRule> rules = new ArrayList<>();
    
    /**
     * 初始化默认规则
     */
    @PostConstruct
    public void initializeRules() {
        // 数据隐私规则
        rules.add(new DataPrivacyRule());
        
        // 金融合规规则
        rules.add(new FinancialComplianceRule());
        
        // 内容安全规则
        rules.add(new ContentSafetyRule());
        
        // 操作权限规则
        rules.add(new PermissionRule());
        
        // 审计日志规则
        rules.add(new AuditRule());
    }
    
    /**
     * 执行合规检查
     */
    public ComplianceReport check(Action action) {
        ComplianceReport report = new ComplianceReport();
        
        for (ComplianceRule rule : rules) {
            RuleResult result = rule.evaluate(action);
            report.addResult(result);
            
            if (result.getSeverity() == Severity.CRITICAL) {
                // 记录违规
                auditLog.recordViolation(action, result);
                
                // 通知治理委员会
                notifyGovernanceBoard(action, result);
                
                // 阻断操作
                throw new ComplianceViolationException(result.getMessage());
            }
        }
        
        return report;
    }
}

/**
 * 数据隐私规则
 */
public class DataPrivacyRule implements ComplianceRule {
    
    @Override
    public RuleResult evaluate(Action action) {
        // 检查是否涉及个人数据
        if (action.getDataType() == DataType.PERSONAL) {
            // 检查是否有合法依据
            if (!action.hasLegalBasis()) {
                return RuleResult.fail("处理个人数据必须具有合法依据");
            }
            
            // 检查是否超出目的范围
            if (!action.isWithinPurpose()) {
                return RuleResult.fail("数据处理超出声明目的");
            }
            
            // 检查数据最小化原则
            if (action.isOverCollection()) {
                return RuleResult.warn("收集的数据超出必要范围");
            }
        }
        
        return RuleResult.pass();
    }
}

/**
 * 金融合规规则
 */
public class FinancialComplianceRule implements ComplianceRule {
    
    private static final BigDecimal SINGLE_LIMIT = new BigDecimal("100000");
    private static final BigDecimal DAILY_LIMIT = new BigDecimal("500000");
    
    @Override
    public RuleResult evaluate(Action action) {
        if (action.getType() == ActionType.FINANCIAL) {
            BigDecimal amount = action.getAmount();
            
            // 检查单笔限额
            if (amount.compareTo(SINGLE_LIMIT) > 0) {
                return RuleResult.block("单笔交易超过10万元，需要人工审批");
            }
            
            // 检查日累计限额
            BigDecimal dailyTotal = getDailyTotal(action.getUserId());
            if (dailyTotal.add(amount).compareTo(DAILY_LIMIT) > 0) {
                return RuleResult.block("日累计交易超过50万元，需要人工审批");
            }
            
            // 检查反洗钱规则
            if (isSuspicious(action)) {
                return RuleResult.block("触发反洗钱规则，需要人工审核");
            }
        }
        
        return RuleResult.pass();
    }
}
```

```java
/**
 * 分级审批管理器

@Component
public class ApprovalManager {
    
    @Autowired
    private HumanTaskService humanTaskService;
    
    /**
     * 根据风险等级确定审批流程
     */
    public ApprovalWorkflow determineWorkflow(Action action) {
        RiskLevel risk = assessRisk(action);
        
        return switch (risk) {
            case LOW -> new AutoApprovalWorkflow();  // 自动通过
            case MEDIUM -> new SingleApprovalWorkflow();  // 一级审批
            case HIGH -> new MultiApprovalWorkflow();  // 多级审批
            case CRITICAL -> new BoardApprovalWorkflow();  // 委员会审批
        };
    }
    
    /**
     * 提交人工审批
     */
    public ApprovalTicket submitForApproval(Action action, String reason) {
        ApprovalWorkflow workflow = determineWorkflow(action);
        
        ApprovalTicket ticket = ApprovalTicket.builder()
            .action(action)
            .workflow(workflow)
            .submittedAt(Instant.now())
            .status(ApprovalStatus.PENDING)
            .build();
        
        // 发送通知给审批人
        workflow.getApprovers().forEach(approver -
            notificationService.notify(approver, ticket));
        
        return ticket;
    }
    
    /**
     * 审批超时处理
     */
    @Scheduled(fixedRate = 60000)
    public void handleTimeout() {
        List<ApprovalTicket> overdue = ticketRepo.findOverdue(Duration.ofHours(24));
        
        for (ApprovalTicket ticket : overdue) {
            // 升级给上级
            escalate(ticket);
            
            // 发送提醒
            sendReminder(ticket);
        }
    }
}
```

```java
/**
 * 审计日志服务
 */
@Service
public class AuditService {
    
    @Autowired
    private AuditLogRepository auditRepo;
    
    /**
     * 记录操作日志
     */
    public void record(Action action, Result result) {
        AuditLog log = AuditLog.builder()
            .id(UUID.randomUUID().toString())
            .timestamp(Instant.now())
            .agentId(action.getAgentId())
            .actionType(action.getType())
            .input(maskSensitiveData(action.getInput()))
            .output(maskSensitiveData(result.getOutput()))
            .decision(result.getDecision())
            .reason(result.getReason())
            .complianceReport(result.getComplianceReport())
            .build();
        
        // 写入不可篡改存储
        auditRepo.save(log);
        
        // 同步到区块链（可选）
        blockchain.record(log.getHash());
    }
    
    /**
     * 生成审计报告
     */
    public AuditReport generateReport(TimeRange range) {
        List<AuditLog> logs = auditRepo.findByTimeRange(range);
        
        return AuditReport.builder()
            .totalActions(logs.size())
            .complianceRate(calculateComplianceRate(logs))
            .violationCount(countViolations(logs))
            .highRiskActions(findHighRiskActions(logs))
            .trendAnalysis(analyzeTrend(logs))
            .build();
    }
    
    /**
     * 数据脱敏
     */
    private String maskSensitiveData(String data) {
        // 身份证号脱敏
        data = data.replaceAll("(\\d{6})\\d{8}(\\d{4})", "$1********$2");
        
        // 手机号脱敏
        data = data.replaceAll("(\\d{3})\\d{4}(\\d{4})", "$1****$2");
        
        // 银行卡号脱敏
        data = data.replaceAll("(\\d{4})\\d{8,12}(\\d{4})", "$1 **** **** $2");
        
        return data;
    }
}
```

```java
@Component
public class AnomalyDetectionAgent {
    
    @Autowired
    private MLService mlService;
    
    /**
     * 实时异常检测
     */
    public void analyze(AgentActivity activity) {
        // 1. 行为基线
        BehaviorProfile baseline = getBaseline(activity.getAgentId());
        
        // 2. 异常指标
        List<AnomalyIndicator> indicators = new ArrayList<>();
        
        // 频率异常
        if (activity.getFrequency() > baseline.getAvgFrequency() * 3) {
            indicators.add(new AnomalyIndicator("FREQUENCY", "操作频率异常增高"));
        }
        
        // 时间异常
        if (isUnusualTime(activity.getTimestamp())) {
            indicators.add(new AnomalyIndicator("TIMING", "非常规时间操作"));
        }
        
        // 范围异常
        if (activity.getScope() > baseline.getAvgScope() * 2) {
            indicators.add(new AnomalyIndicator("SCOPE", "操作范围异常扩大"));
        }
        
        // 3. 综合评估
        if (!indicators.isEmpty()) {
            RiskScore score = calculateRiskScore(indicators);
            
            if (score.getValue() > 80) {
                // 高风险，立即阻断
                blockAgent(activity.getAgentId());
                alertService.sendCriticalAlert(activity, indicators);
            } else if (score.getValue() > 50) {
                // 中风险，加强监控
                enhanceMonitoring(activity.getAgentId());
                alertService.sendWarning(activity, indicators);
            }
        }
    }
}
```

```java
@Component
public class PrivacyProtectionService {
    
    /**
     * 差分隐私处理
     */
    public Dataset addNoise(Dataset dataset, double epsilon) {
        return dataset.map(row -> {
            // 添加拉普拉斯噪声
            double noise = laplaceNoise(epsilon);
            return row.addNoise(noise);
        });
    }
    
    /**
     * 联邦学习
     */
    public Model federatedLearn(List<DataNode> nodes) {
        // 1. 初始化全局模型
        Model globalModel = initializeModel();
        
        // 2. 多轮迭代
        for (int round = 0; round < MAX_ROUNDS; round++) {
            List<ModelUpdate> updates = new ArrayList<>();
            
            for (DataNode node : nodes) {
                // 本地训练
                Model localModel = node.trainLocal(globalModel);
                
                // 上传模型更新（不上传原始数据）
                updates.add(localModel.getUpdate());
            }
            
            // 聚合更新
            globalModel.aggregate(updates);
        }
        
        return globalModel;
    }
    
    /**
     * 数据脱敏
     */
    public String anonymize(String data, AnonymizationType type) {
        return switch (type) {
            case MASKING -> mask(data);
            case TOKENIZATION -> tokenize(data);
            case GENERALIZATION -> generalize(data);
            case SUPPRESSION -> "[REDACTED]";
        };
    }
}
```

```java
@Service
public class CreditCardApprovalGovernance {
    
    @Autowired
    private ComplianceEngine complianceEngine;
    
    @Autowired
    private ApprovalManager approvalManager;
    
    @Autowired
    private AuditService auditService;
    
    /**
     * 审批流程（带治理）
     */
    public ApprovalResult approve(Application application) {
        // 1. 输入校验
        validateInput(application);
        
        // 2. 合规检查
        ComplianceReport compliance = complianceEngine.check(
            Action.builder()
                .type(ActionType.CREDIT_APPROVAL)
                .data(application)
                .build()
        );
        
        if (!compliance.isPassed()) {
            return ApprovalResult.builder()
                .status(ApprovalStatus.REJECTED)
                .reason("合规检查未通过：" + compliance.getViolations())
                .build();
        }
        
        // 3. AI 自动审批
        AIResult aiResult = aiModel.evaluate(application);
        
        // 4. 判断是否需要人工复核
        if (needsHumanReview(application, aiResult)) {
            ApprovalTicket ticket = approvalManager.submitForApproval(
                Action.builder()
                    .type(ActionType.CREDIT_APPROVAL)
                    .data(application)
                    .aiResult(aiResult)
                    .build(),
                "高风险申请，需要人工复核"
            );
            
            return ApprovalResult.builder()
                .status(ApprovalStatus.PENDING_REVIEW)
                .ticketId(ticket.getId())
                .build();
        }
        
        // 5. 自动审批通过
        ApprovalResult result = ApprovalResult.builder()
            .status(ApprovalStatus.APPROVED)
            .aiResult(aiResult)
            .build();
        
        // 6. 记录审计日志
        auditService.record(
            Action.builder()
                .type(ActionType.CREDIT_APPROVAL)
                .data(application)
                .build(),
            result
        );
        
        return result;
    }
    
    /**
     * 判断是否需要人工复核
     */
    private boolean needsHumanReview(Application app, AIResult aiResult) {
        // 高额申请
        if (app.getRequestedLimit() > 100000) return true;
        
        // AI 置信度低
        if (aiResult.getConfidence() < 0.8) return true;
        
        // 风险评分高
        if (aiResult.getRiskScore() > 70) return true;
        
        // 敏感人群
        if (app.isSensitiveOccupation()) return true;
        
        // 异常申请模式
        if (app.isAbnormalPattern()) return true;
        
        return false;
    }
}
```

## 常见陷阱与最佳实践

### 陷阱1：治理过度导致效率低下

```markdown
❌ 错误：
所有操作都需要人工审批

✅ 正确：
- 低风险操作自动审批
- 中风险操作抽样审批
- 高风险操作强制审批
- 设置合理的审批时限
```

### 陷阱2：事后审计代替事前预防

```markdown
❌ 错误：
只关注事后追责，忽视事前预防

✅ 正确：
- 事前：权限控制、输入验证
- 事中：实时监控、异常检测
- 事后：审计追溯、复盘改进
```

### 陷阱3：忽视用户体验

```markdown
❌ 错误：
为了安全牺牲用户体验（如频繁二次验证）

✅ 正确：
- 基于风险自适应验证
- 低风险场景免验证
- 高风险场景加强验证
- 平衡安全与便利
```

## 面试题与参考答案

### 1. AI 治理的核心原则是什么？

**答案：**
- 透明性：AI 决策可解释
- 公平性：避免偏见和歧视
- 安全性：防止滥用和攻击
- 隐私性：保护用户数据
- 问责性：明确责任归属

### 2. 如何平衡 AI 自动化与人工监督？

**答案：**
- 分级管理：低风险自动，高风险人工
- 置信度阈值：AI 置信度高则自动，低则人工
- 异常处理：异常情况自动转人工
- 抽样审核：定期抽样人工复核

### 3. 审计日志需要记录哪些内容？

**答案：**
- 操作主体（谁）
- 操作对象（什么）
- 操作时间（何时）
- 操作内容（做了什么）
- 操作结果（结果如何）
- 操作原因（为什么）
- 审批链（谁批准的）

### 4. 如何实现 AI 决策的可解释性？

**答案：**
- 使用可解释模型（决策树、线性模型）
- 特征重要性分析
- SHAP/LIME 解释技术
- 决策链记录
- 可视化展示

### 5. 数据隐私保护的常用技术？

**答案：**
- 数据脱敏（掩码、泛化）
- 差分隐私
- 联邦学习
- 同态加密
- 安全多方计算

## 小结

AI 治理的核心目标：**让 AI 可信、可控、可用**

1. **可信**：决策透明、公平、可解释
2. **可控**：权限管理、人工监督、熔断机制
3. **可用**：不影响业务效率、用户体验良好

Mindra 治理优势：
- 内置三层治理架构
- 自动化合规检查
- 完整审计追踪
- 人工监督无缝集成
